const artifacts = [];
const clean = text => text.replaceAll(root, '<workspace>').replaceAll(process.env.HOME, '<home>').replace(/\b[A-Z0-9]{5}(?:-[A-Z0-9]{5}){4}\b/g, '<disposable-factor-redacted>');
function cleanReport(value) {
  if (!value || typeof value !== 'object') return value;
  if (Array.isArray(value)) return value.map(cleanReport);
  const result = Object.fromEntries(Object.entries(value).map(([key, item]) => [key, typeof item === 'string' ? clean(item) : cleanReport(item)]));
  if (typeof value.body === 'string' && /^(application\/json|text\/)/.test(value.contentType ?? '')) {
    result.body = Buffer.from(clean(Buffer.from(value.body, 'base64').toString())).toString('base64');
  }
  return result;
}
function retain(file, name, compressed = false, report = false) {
  const original = fs.readFileSync(file);
  const normalized = Buffer.from(report ? JSON.stringify(cleanReport(JSON.parse(original)), null, 2) + '\n' : clean(original.toString()));
  const bytes = compressed ? gzipSync(normalized, { level: 9 }) : normalized;
  const dest = path.join(target, name);
  fs.mkdirSync(path.dirname(dest), { recursive: true });
  fs.writeFileSync(dest, bytes);
  artifacts.push({ path: name, originalSha256: hash(original), normalizedSha256: hash(normalized), sha256: hash(bytes), bytes: bytes.length });
}
function json(name, value) {
  const file = path.join(input, name.replaceAll('/', '-'));
  fs.writeFileSync(file, JSON.stringify(value, null, 2) + '\n');
  retain(file, name);
}
