  for (const path of files) mkdirSync(dirname(join(root, path)), { recursive: true });
  for (const path of files.filter(
    (p) => p.startsWith('scripts/') || p.endsWith('browser-fixture.ts'),
  ))
    copyFileSync(path, join(root, path));
  writeFileSync(join(root, 'package.json'), '{"type":"module"}');

  const first = await observed.run(observed.stage);

    if (!statSync(full).isFile() || !/\.(?:json|jsonl|log)$/.test(file) || /(?:browser-profile|profile-data)/.test(file)) continue;

    files.push({path:file,text:readFileSync(full,'utf8')});
  }
  const observation = { source: "<test-checkout>/test/release-validation/playwright-native.test.mjs", sourceSha256: "daa8326634bce206f330c13f677185ecdad18c221b59c1f264c5b89cb77ed5dd", method: 'Verbatim existing fixture constructor; same 15-second stage limit; first stage observed outside node:test.
