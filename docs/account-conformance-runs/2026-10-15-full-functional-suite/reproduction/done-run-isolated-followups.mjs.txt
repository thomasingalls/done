const results = [];
const saveResults = () => fs.writeFileSync(path.join(evidence, 'isolated-followup-results.json'), JSON.stringify(results, null, 2) + '\n');
async function run(command, cwd, label) {
  const log = 'followup-' + label + '.log';
  const fd = fs.openSync(path.join(evidence, log), 'wx');
  const row = { command, cwd, log, startedAt: new Date().toISOString(), status: 'running' };
  results.push(row); saveResults(); console.log('START ' + command);
  const env = { ...process.env, PATH: '/private/tmp/done-pnpm11/bin:' + path.join(cwd, 'node_modules/.bin') + ':' + process.env.PATH, pnpm_config_verify_deps_before_run: 'warn', NO_UPDATE_NOTIFIER: '1' };
  const child = spawn('/bin/sh', ['-c', command], { cwd, env, stdio: ['ignore', fd, fd], detached: true });
  row.pid = child.pid; saveResults();
  row.exitCode = await new Promise(resolve => { child.once('error', error => { row.error = String(error); resolve(null); }); child.once('exit', (code, signal) => { row.signal = signal; resolve(code); }); });
  fs.closeSync(fd); row.status = row.exitCode === 0 ? 'passed' : 'failed'; row.completedAt = new Date().toISOString(); saveResults(); console.log('END ' + row.status + ' ' + command);
  return row;
}
