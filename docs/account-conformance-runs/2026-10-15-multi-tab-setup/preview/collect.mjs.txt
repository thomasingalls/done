const hash = b => crypto.createHash('sha256').update(b).digest('hex');
const normalized = text => text.replaceAll(root, '<workspace>').replaceAll(process.env.HOME, '<home>')

const artifacts = [];
function retain(file, name, compress = false) {
  const original = fs.readFileSync(file);
  const clean = Buffer.from(normalized(original.toString()));
  const bytes = compress ? gzipSync(clean, { level: 9 }) : clean;
  fs.writeFileSync(path.join(target, name), bytes);
  artifacts.push({ path: name, source: path.basename(file), originalSha256: hash(original), normalizedSha256: hash(clean), retainedSha256: hash(bytes), retainedBytes: bytes.length });
}
