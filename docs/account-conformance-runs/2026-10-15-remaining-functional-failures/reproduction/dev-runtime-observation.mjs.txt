import fs from 'node:fs';
import path from 'node:path';
import { createRequire } from 'node:module';
import { pathToFileURL } from 'node:url';
const root = '<workspace>';
const target = '/private/tmp/done-remaining-suite-6eeb7d5f';
const require = createRequire(path.join(root, 'package.json'));
const { createServer } = await import(pathToFileURL(require.resolve('vite')).href);
const playwright = await import(pathToFileURL(require.resolve('@playwright/test')).href);
const chromium = playwright.chromium ?? playwright.default.chromium;
const results = { source: '6eeb7d5f', cases: [] };
const save = () => fs.writeFileSync(path.join(target, 'dev-runtime-observation.json'), JSON.stringify(results, null, 2) + '\n');
const browser = await chromium.launch({ headless: true });
try {
  for (const needsInterop of [false, true]) {
    const result = { needsInterop, freshCache: true, errors: [], moduleExports: [], shell: 0 };
    results.cases.push(result);
    save();
    const server = await createServer({
      configFile: path.join(root, 'vite/privateMigration.config.ts'),
      root,
      cacheDir: path.join(target, needsInterop ? 'cache-interop' : 'cache-current'),
      optimizeDeps: { force: true, ...(needsInterop ? { needsInterop: ['react/compiler-runtime'] } : {}) },
      server: { host: '127.0.0.1', port: 0, open: false },
      logLevel: 'error',
    });
    const context = await browser.newContext();
    try {
      await server.listen();
      const origin = server.resolvedUrls.local[0];
      const page = await context.newPage();
      const captures = [];
      page.on('pageerror', error => result.errors.push(error.message));
      page.on('response', response => {
        if (response.url().includes('react_compiler-runtime.js')) {
          captures.push(response.text().then(text => {
            result.moduleExports.push(...text.split('\n').filter(line => /export.*\{|default:|c:/.test(line)).slice(-12));
          }).catch(error => result.errors.push(error.message)));
        }
      });
      await page.goto(new URL('local.html', origin).href);
      await page.waitForSelector('.shell, .local-startup', { timeout: 10000 }).catch(error => result.errors.push(error.message));
      await Promise.all(captures);
      result.shell = await page.locator('.shell').count();
      result.startup = await page.locator('.local-startup').count();
      result.editable = await page.locator('[contenteditable="true"]').count();
      save();
    } finally {
      await context.close();
      await server.close();
    }
  }
} finally {
  await browser.close();
  save();
}
console.log(JSON.stringify(results));
