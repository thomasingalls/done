function retain(file, name, compressed = false) {
  const original = fs.readFileSync(file);
  const clean = Buffer.from(normalize(original));
  const bytes = compressed ? gzipSync(clean, { level: 9 }) : clean;
  const dest = path.join(target, name);
  fs.mkdirSync(path.dirname(dest), { recursive: true });
  fs.writeFileSync(dest, bytes);
  artifacts.push({ path: name, originalSha256: sha(original), normalizedSha256: sha(clean), sha256: sha(bytes), bytes: bytes.length });
}
