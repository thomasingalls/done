import test from'node:test';import assert from'node:assert/strict';import{createHash}from'node:crypto';import{projectRequest,projectResponse}from'./frame-probe.mjs';
import{encodeSubmit}from'./source/account-protocol/deliveryTransport.ts';import{encodeSubmission}from'./source/account-protocol/submission.ts';import{encodeNetworkEnvelope}from'./source/account-protocol/envelope.ts';import{encodePositionResponse}from'./source/account-protocol/positionDecision.ts';import{encodeInboundRequest,encodeInboundPage}from'./source/account-protocol/inboundTransport.ts';import{encodeInboundBatch,encodeInboundBatchPages}from'./source/account-protocol/inboundBatch.ts';
const sha=x=>createHash('sha256').update(x).digest('hex'),z=n=>'00'.repeat(n),workspace=z(16),publisher='11'.repeat(16),sig=z(64),read=(after=37,pub=publisher)=>encodeInboundRequest({serviceId:z(16),origin:'http://localhost:53781',workspace,reader:'22'.repeat(16),publisher:pub,lineage:z(32),head:z(32),epoch:3,after,limit:1,nonce:z(32),expires:10000,signature:sig});
const env=()=>encodeNetworkEnvelope({v:2,workspace_id:workspace,credential_id:publisher,epoch:3,gen:1,seq:38,nonce:new Uint8Array(12),digest:new Uint8Array(32),ct:new TextEncoder().encode('PRIVATE_TEST_CONTENT_NEVER_LOG'),sig:new Uint8Array(64)});
const proof=()=>encodePositionResponse({kind:'accepted',head:{workspace,credential:publisher,position:38,link:z(32),epoch:3,final:false,signature:sig},leaf:{epoch:3,digest:z(32)}});
const page=(empty=false)=>encodeInboundPage({request:sha(read()),through:38,next:38,entries:empty?[]:[{envelope:env(),proof:proof()}],signature:sig});
test('frame projection retains exact submission publisher and position without ciphertext',()=>{const bytes=encodeSubmit(workspace,encodeSubmission({lineage:z(32),envelope:env(),parent:null,signature:new Uint8Array(64)})),v=projectRequest('/account/delivery/submit',bytes);assert.equal(v.position,38);assert.equal(v.publisherSha256,sha(publisher));assert.equal(v.requestSha256,sha(bytes));assert.equal(v.unverified,true);assert(!JSON.stringify(v).includes('PRIVATE_TEST_CONTENT_NEVER_LOG'))});
test('frame projection keeps exact per-request batch identities and future cursors',()=>{const v=projectRequest('/account/delivery/read-v2-batch',encodeInboundBatch([read(),read(38)]));assert.deepEqual(v.requests.map(x=>x.after),[37,38]);assert.notEqual(v.requests[0].requestSha256,v.requests[1].requestSha256);assert.equal(v.requests[0].limit,1)});
test('frame projection distinguishes publishers and does not expose raw credential IDs',()=>{const v=projectRequest('/account/delivery/read-v2-batch',encodeInboundBatch([read(),read(37,'33'.repeat(16))]));assert.notEqual(v.requests[0].publisherSha256,v.requests[1].publisherSha256);assert(!JSON.stringify(v).includes(publisher))});
test('frame projection pairs a returned entry with its original request and accepted position',()=>{const v=projectResponse('/account/delivery/read-v2',page());assert.equal(v.pages[0].requestSha256,sha(read()));assert.equal(v.pages[0].entries[0].envelope.position,38);assert.equal(v.pages[0].entries[0].decision.position,38);assert.equal(v.pages[0].through,38);assert.equal(v.pages[0].next,38);assert.equal(v.unverified,true)});
test('frame projection retains empty and bounded multi-page responses without inventing entries',()=>{const v=projectResponse('/account/delivery/read-v2-batch',encodeInboundBatchPages([page(),page(true)]));assert.deepEqual(v.pages.map(p=>p.entries.length),[1,0])});
test('frame projection marks an accepted-shaped response unverified rather than asserting trust',()=>{const v=projectResponse('/account/delivery/submit',proof());assert.equal(v.kind,'accepted');assert.equal(v.unverified,true);assert(!('verified'in v));assert(!('signature'in v));assert(!('credential'in v))});
test('frame projection preserves missing bodies as missing evidence',()=>{assert.deepEqual(projectResponse('/account/delivery/submit',null),{unverified:true,missing:true});assert.deepEqual(projectRequest('/account/delivery/read-v2',undefined),{unverified:true,missing:true})});
test('frame projection rejects malformed, noncanonical and oversized request bytes',()=>{assert.throws(()=>projectRequest('/account/delivery/read-v2',new Uint8Array(5000)));assert.throws(()=>projectRequest('/account/delivery/read-v2',new TextEncoder().encode(JSON.stringify(JSON.parse(new TextDecoder().decode(read())),null,1))));assert.throws(()=>projectResponse('/account/delivery/read-v2',new Uint8Array([1,2])))});
test('frame projection leaves unrelated routes outside its metadata scope',()=>{assert.equal(projectRequest('/account/history/v1/checkpoint',new Uint8Array()),undefined);assert.equal(projectResponse('/account/history/v1/checkpoint',new Uint8Array()),undefined)});

test('frame projection never starts body observation for unrelated streaming or bodyless routes',()=>{assert.equal(projectRequest('/account/notifications',null),undefined);assert.equal(projectResponse('/account/notifications',null),undefined)});
