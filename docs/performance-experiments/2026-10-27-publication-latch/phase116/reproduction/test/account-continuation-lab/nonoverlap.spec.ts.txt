import{readRequestInfo}from'../../../read-request.mjs';
import{test,expect,chromium}from'@playwright/test';
import{readFileSync,writeFileSync,mkdirSync,cpSync}from'node:fs';import{join}from'node:path';
import{start,stop,control}from'./runtime.mjs';import{restoreAuthenticator,openSaved}from'./devices';import{seedProfile}from'./seed-profile';
import{root,card,focus,position,exported,archiveIdentity,lock,digest}from'./common';

const out=process.env['FAIR_OUT']!,exe=process.env['CHROMIUM_PATH']!;
test('accepted writing survives non-overlapping H sessions in both directions',async()=>{
 mkdirSync(out,{recursive:true,mode:0o700});const fixture=JSON.parse(readFileSync(root+'/private-fixture/fixture.json','utf8')),a=fixture.accounts.find((x:any)=>x.label==='H');let ctx:any;const result:any={variant:process.env['FAIR_VARIANT'],outcome:'starting',transitions:[],network:[],roots:[]};
 const open=async(index:number)=>{if(ctx)throw Error('Sender must be closed before the receiver opens');const saved=a.actors[index];const path=join(out,'private-profile-H'+index);ctx=await chromium.launchPersistentContext(path,{executablePath:exe,baseURL:'http://localhost:53781',viewport:{width:1440,height:1000}});result.roots.push(await browserIdentity(ctx.browser()));await seedProfile(ctx,saved.storage);const p=ctx.pages()[0]??await ctx.newPage();p.on('response',(r:any)=>{const path=new URL(r.url()).pathname;if(path.startsWith('/account/'))result.network.push({device:index,path,status:r.status(),at:Date.now(),...readRequestInfo(r.request())})});await restoreAuthenticator(p,saved.credentials);await p.goto('/');await p.evaluate(()=>navigator.serviceWorker.ready);await p.reload();await p.waitForFunction(()=>!!navigator.serviceWorker.controller);const worker=await p.evaluate(async()=>{const r=await navigator.serviceWorker.ready;return{script:r.active!.scriptURL,state:r.active!.state,bytes:Array.from(new Uint8Array(await(await fetch(r.active!.scriptURL)).arrayBuffer()))}});expect(digest(Buffer.from(worker.bytes))).toBe(digest(readFileSync(process.env['FAIR_DIST']!+'/sw.js')));result.transitions.push({device:index,openedAt:Date.now(),selectedWorkerSha256:digest(Buffer.from(worker.bytes)),active:true});await ctx.setOffline(true);const at=Date.now();await openSaved(p,saved.code,index>0);const editReadyAt=Date.now();await p.getByLabel('Go to date',{exact:true}).fill(a.date);await expect(card(p,'H')).toBeVisible();result.transitions.at(-1).offlineUnlockToEditMs=editReadyAt-at;return p;};
 const close=async(p:any)=>{await lock(p);await ctx.close();ctx=undefined;result.transitions.at(-1).closedAt=Date.now();};
 try{
  const state=join(out,'state');cpSync(fixture.state,state,{recursive:true,errorOnExist:true});await start(state,process.env['FAIR_DIST']!);await control('start');
  let p=await open(1);await expect(card(p,'H')).not.toContainText(a.expected);result.transitions[0].reconnectAt=Date.now();await ctx.setOffline(false);await expect(card(p,'H')).toContainText(a.expected,{timeout:600000});result.transitions[0].caughtUpAt=Date.now();expect(await exported(p)).toEqual(a.archive);result.transitions[0].exactArchive=true;
  const before=await position(p),marker=' Non-overlap accepted reverse 20261027;';await focus(p,'H',true);await p.keyboard.insertText(marker);await p.keyboard.press('Escape');await expect.poll(()=>position(p),{timeout:30000}).toBeGreaterThan(before);const reverse=await exported(p);result.transitions[0].confirmedAcceptedAt=Date.now();result.transitions[0].acceptedBefore=before;result.transitions[0].acceptedAfter=await position(p);result.transitions[0].archive=archiveIdentity(reverse);await close(p);
  p=await open(0);expect(result.transitions[0].closedAt).toBeLessThan(result.transitions[1].openedAt);await expect(card(p,'H',true)).not.toContainText(marker);result.transitions[1].reconnectAt=Date.now();await ctx.setOffline(false);await expect(card(p,'H',true)).toContainText(marker,{timeout:120000});result.transitions[1].caughtUpAt=Date.now();expect(await exported(p)).toEqual(reverse);result.transitions[1].exactArchive=true;
  await ctx.setOffline(true);await lock(p);await ctx.close();ctx=undefined;const path=join(out,'private-profile-H0');ctx=await chromium.launchPersistentContext(path,{executablePath:exe,baseURL:'http://localhost:53781',viewport:{width:1440,height:1000},offline:true});p=ctx.pages()[0]??await ctx.newPage();await restoreAuthenticator(p,a.actors[0].credentials);await p.goto('/');await openSaved(p,a.actors[0].code);await p.getByLabel('Go to date',{exact:true}).fill(a.date);expect(await exported(p)).toEqual(reverse);result.coldSameProfileOfflineExact=true;result.noStorageReinjectionOnColdReopen=true;result.noSimultaneousOpenSenders=true;result.metrics=(await control('end')as any).metrics;result.trace=await control('trace');result.outcome='passed';
 }catch(e){result.outcome='failed';result.error=String(e);throw e}finally{await ctx?.close().catch(()=>{});writeFileSync(join(out,'result.json'),JSON.stringify(result,null,2)+'\n');await stop();}
});
