// Design sketch only: not exported to, imported by or executed in the product.
class RecheckSketch {
  constructor(ttl = 1000) { this.ttl = ttl; this.generation = {}; this.quiet = new Map(); }
  capture() { return this.generation; }
  invalidate() { this.generation = {}; this.quiet.clear(); }
  empty(publisher, basis, capture, now) {
    if (capture !== this.generation) return false;
    if (!this.quiet.has(publisher) && this.quiet.size >= 128) this.quiet.delete(this.quiet.keys().next().value);
    this.quiet.set(publisher, { basis, at: now }); return true;
  }
  defer(publisher, basis, now, { backlog = true, self = false, conflicted = false } = {}) {
    const q = this.quiet.get(publisher);
    return !!q && backlog && !self && !conflicted && q.basis === basis && now >= q.at && now - q.at < this.ttl;
  }
  productive(publisher) { this.quiet.delete(publisher); }
}
