const marker = ' Quiet peer arrival 20261027;';
function markerProbe(value: string) {
  const now = () => performance.timeOrigin + performance.now();
  const result: any = { markerAt: null, inputs: [] };
  const observe = () => {
    if (result.markerAt !== null) return;
    const c = [...document.querySelectorAll('.day-panel[data-active="true"] .card')].find(c => c.textContent?.includes('Fair H reverse:'));
    if (c?.textContent?.includes(value)) result.markerAt = now();
  };
  new MutationObserver(observe).observe(document, { subtree: true, childList: true, characterData: true });
  addEventListener('beforeinput', (event: Event) => {
    if (!(event.target instanceof Element) || !event.target.closest('.card')) return;
    result.inputs.push({ at: now(), trusted: event.isTrusted, focused: document.hasFocus(), visible: document.visibilityState });
  }, true);
  (window as any).__quietMarker = result;
}
