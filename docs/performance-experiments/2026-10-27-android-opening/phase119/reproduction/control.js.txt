(() => {
  if (document.querySelector('.cm-content[contenteditable="true"]')) return 'Lock first';
  if (globalThis.__doneOpeningControl) return 'Control already exists';
  const started = performance.now(), events = [], reads = [], tasks = [], restore = [];
  let active = true, timer;
  const safe = (fn) => { try { return fn(); } catch {} };
  const now = () => Math.round((performance.now() - started) * 10) / 10;
  const mark = (kind) => { if (events.length < 64) events.push({kind, at: now(), visible: document.visibilityState === 'visible'}); };
  const credential = navigator.credentials, originalGet = credential.get;
  const getDescriptor = Object.getOwnPropertyDescriptor(credential, 'get');
  const get = function (...args) {
    mark('providerRequested');
    const result = Reflect.apply(originalGet, this, args);
    safe(() => result.then(() => mark('providerReturned'), () => mark('providerRejected')));
    return result;
  };
  Object.defineProperty(credential, 'get', {configurable: true, writable: true, value: get});
  restore.push(() => { if (credential.get === get) { if (getDescriptor) Object.defineProperty(credential, 'get', getDescriptor); else delete credential.get; } });
  const proto = IDBObjectStore.prototype, originalAll = proto.getAll;
  const allDescriptor = Object.getOwnPropertyDescriptor(proto, 'getAll');
  const all = function (...args) {
    const row = {at: now(), store: ['content','meta','heads','outbound'].includes(this.name) ? this.name : 'other', frames: []};
    safe(() => { row.frames = (new Error().stack ?? '').split('\n').flatMap(line => line.match(/https:\/\/done\.oxidationlabs\.com\/assets\/[A-Za-z0-9_.-]+\.js:\d+:\d+/g) ?? []).slice(0, 8); });
    if (reads.length < 32) reads.push(row);
    const result = Reflect.apply(originalAll, this, args);
    safe(() => result.addEventListener('success', () => safe(() => { row.completedAt = now(); row.rows = Array.isArray(result.result) ? result.result.length : 0; }), {once: true}));
    return result;
  };
  Object.defineProperty(proto, 'getAll', {...allDescriptor, value: all});
  restore.push(() => { if (proto.getAll === all) Object.defineProperty(proto, 'getAll', allDescriptor); });
  const onVisibility = () => mark('visibilityChanged');
  document.addEventListener('visibilitychange', onVisibility);
  let po;
  safe(() => { po = new PerformanceObserver(list => { for (const e of list.getEntries()) if (tasks.length < 64) tasks.push({at: Math.round((e.startTime - started) * 10) / 10, duration: e.duration}); }); po.observe({type:'longtask',buffered:false}); });
  const mo = new MutationObserver(() => safe(() => { if (active && document.querySelector('.cm-content[contenteditable="true"]')) { mark('editorEditable'); stop('editor'); } }));
  mo.observe(document.getElementById('root') ?? document.body, {childList:true,subtree:true,attributes:true,attributeFilter:['contenteditable']});
  const read = () => ({version:1,active,elapsedMs:now(),events,reads,longTasks:tasks,scope:'No CPU sampling/per-crypto wrappers. Numeric rows and public script locations only; no values or arguments.'});
  function stop(reason='manual') {
    if (!active) return read();
    mark(reason === 'editor' ? 'stoppedOnEditor' : reason === 'timeout' ? 'timedOut' : 'manualStop');
    active=false; clearInterval(timer); mo.disconnect(); po?.disconnect(); document.removeEventListener('visibilitychange',onVisibility);
    for (const fn of restore.reverse()) safe(fn);
    return read();
  }
  globalThis.__doneOpeningControl = Object.freeze({read,stop});
  timer=setInterval(() => {if(performance.now()-started>=300000)stop('timeout');},1000);
  mark('armed');
  return 'Light confirming probe armed; no CPU sampling or per-crypto observers.';
})();
