async function offer(cdp:CDPSession,text:string,period:number,result:any,label:string){
 const began=now(), commands:Promise<unknown>[]=[];
 for(let i=0;i<text.length;i++){
  const planned=began+i*period; const wait=planned-now(); if(wait>0)await pause(wait);
  const row:any={label,index:i,planned,offered:now(),done:0};result.offers.push(row);
  const ch=text[i]!;
  const down=cdp.send('Input.dispatchKeyEvent',{type:'keyDown',key:ch,text:ch,unmodifiedText:ch});
  const up=cdp.send('Input.dispatchKeyEvent',{type:'keyUp',key:ch});
  commands.push(Promise.all([down,up]).then(()=>{row.done=now();}));
 }
 await Promise.all(commands);
}
