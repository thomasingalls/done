import { REQUEST_LIMITS } from './requestBudget.ts';

/** Scheduling only: no authority, response or verification-result reuse.
 * The handler lends its bounded immutable body until its existing operation ends. */
export function createBatchReadAdmission() {
  const active = new Set<{ scope: string; body: Buffer }>();
  return {
    acquire(scope: string, body: Buffer): (() => void) | undefined {
      if (active.size >= REQUEST_LIMITS.active) return;
      for (const entry of active) if (entry.scope === scope && entry.body.length === body.length) return;
      const entry = { scope, body };
      active.add(entry);
      let live = true;
      return () => {
        if (!live) return;
        live = false;
        active.delete(entry);
      };
    },
  };
}
