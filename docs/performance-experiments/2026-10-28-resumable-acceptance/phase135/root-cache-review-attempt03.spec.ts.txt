import{test,expect,chromium}from'@playwright/test';import{readFileSync,writeFileSync,mkdirSync,cpSync}from'node:fs';import{join}from'node:path';
import{root,card,exported,archiveIdentity,lock}from'./common';import{start,stop}from'./runtime.mjs';import{restoreAuthenticator,openSaved}from'./devices';
const base=process.env['FAIR_OUT']!;
for(const tamper of [false,true])test(`real root reopening with ${tamper?'corrupt':'valid'} unfinished auxiliary cache retains ordinary writing`,async()=>{
 const out=base+(tamper?'-corrupt':'-valid');mkdirSync(out,{recursive:true,mode:0o700});process.env['FAIR_OUT']=out;const fixture=JSON.parse(readFileSync(root+'/prepared-fixture/fixture.json','utf8')),a=fixture.accounts.find((a:any)=>a.label==='H'),saved=a.actors[2];let ctx:any;const result:any={kind:'boundary-review',tamper,outcome:'starting'};
 try{
 const profile=join(out,'private-profile');cpSync(root+'/apparatus-pilot-01/private-profile',profile,{recursive:true,errorOnExist:true});const state=join(out,'state');cpSync(root+'/apparatus-pilot-01/state',state,{recursive:true,errorOnExist:true});await start(state,process.env['FAIR_DIST']!);
 ctx=await chromium.launchPersistentContext(profile,{executablePath:process.env['CHROMIUM_PATH']!,baseURL:'http://localhost:53781',viewport:{width:1440,height:1000},offline:true});let p=ctx.pages()[0]??await ctx.newPage();await restoreAuthenticator(p,saved.credentials);await p.goto('/');await expect.poll(async()=>await p.getByLabel('Unlock code',{exact:true}).isVisible() || await p.getByRole('button',{name:/Resume device connection/}).isVisible()).toBe(true);
 result.before=await p.evaluate(async(corrupt:boolean)=>{const caches=(await indexedDB.databases()).filter(d=>/^done\.[0-9a-f]{32}\.preparation-v1$/.test(d.name??''));if(caches.length!==1)throw Error('Expected one unfinished preparation cache');const db=await new Promise<IDBDatabase>((yes,no)=>{const r=indexedDB.open(caches[0]!.name!);r.onsuccess=()=>yes(r.result);r.onerror=()=>no(r.error);});try{return await new Promise((yes,no)=>{const tx=db.transaction(['head','chunks'],corrupt?'readwrite':'readonly'),chunks=tx.objectStore('chunks').count(),h=tx.objectStore('head').get('current');h.onsuccess=()=>{if(corrupt){const r=h.result;r.ct=new Uint8Array(r.ct);r.ct[0]^=1;tx.objectStore('head').put(r,'current');}};tx.oncomplete=()=>yes({chunks:chunks.result,headPresent:!!h.result,corrupted:corrupt});tx.onabort=()=>no(tx.error);});}finally{db.close();}},tamper);
 expect(result.before.chunks).toBeGreaterThan(0);expect(result.before.headPresent).toBe(true);
 await openSaved(p,saved.code,true);await p.getByLabel('Go to date',{exact:true}).fill(a.date);expect(await exported(p)).toEqual(a.archive);await expect(card(p,'H')).toContainText(a.expected);result.archive=archiveIdentity(a.archive);result.ordinaryRootReopen=true;await p.bringToFront();expect(await p.evaluate(()=>document.visibilityState)).toBe('visible');result.readiness=await p.evaluate(()=>({visibility:document.visibilityState,workspace:(window as any).__maintenanceOwner.workspace.status,sync:(window as any).__maintenanceOwner.sync?.status,maintenance:(window as any).__maintenanceRead()}));
 // Let the actual policy inspect/reuse (or discard invalid optional cache) without a manual command.
 await p.waitForFunction(()=>{const r=(window as any).__maintenanceRead?.();return r?.events.some((e:any)=>e.reason==='preparation-saved')||r?.diagnostics.completions>0||r?.status.phase==='halted';},undefined,{timeout:60000});result.maintenance=await p.evaluate(()=>(window as any).__maintenanceRead());expect(result.maintenance.status.phase).not.toBe('halted');expect(result.maintenance.events.some((e:any)=>e.reason==='preparation-saved')||result.maintenance.diagnostics.completions>0).toBe(true);expect(await exported(p)).toEqual(a.archive);
 await lock(p);await ctx.close();ctx=undefined;result.outcome='passed';
 }catch(e){result.outcome='failed';result.error=String(e);throw e;}finally{if(ctx){result.final=await ctx.pages()[0]?.evaluate(()=>({...(window as any).__maintenanceRead?.(),visibility:document.visibilityState,workspace:(window as any).__maintenanceOwner?.workspace.status,sync:(window as any).__maintenanceOwner?.sync?.status})).catch(()=>null);await ctx.close().catch(()=>{});}writeFileSync(join(out,'result.json'),JSON.stringify(result,null,2)+'\n');await stop();}
});
