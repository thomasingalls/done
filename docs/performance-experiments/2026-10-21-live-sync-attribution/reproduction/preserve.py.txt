 source=Path(source);data=source.read_bytes();name=target+('.gz' if compress else '');path=dest/name;path.parent.mkdir(parents=True,exist_ok=True)
 path.write_bytes(gzip.compress(data,mtime=0) if compress else data);manifest[name]={'sourceSha256':sha(data),'storedSha256':sha(path.read_bytes())}
for file in ['source.json','identity.json','production-asset-agreement.json','browser-build.json','protocol.txt','fixture-attempts.txt','fixture-inspection.json','analysis.json','profiles-summary.json','run-primary.log','run-classification.json','evidence-notes.md']:

 put(r/file,'reproduction/'+file+'.txt')
put(r/'diagnostic-instrumentation.patch','reproduction/diagnostic-instrumentation.patch')
put(Path('/private/tmp/done-live-sync-triage-20261021/public-build.json'),'public-build.json')
for file in (r/'source/test/live-sync-attribution').iterdir():
 if file.is_file():put(file,'reproduction/'+file.name+'.txt')
for name in primary+diag+pilots+attempts:

 if log.exists():put(log,'runs/'+name+'.log',True)
 for profile in ['receiver.cpuprofile','server.cpuprofile']:
  if (path/profile).exists():put(path/profile,'profiles/'+name+'-'+profile,True)
 for snapshot in ['probe','probe-final']:
  if (path/snapshot).exists():
   for file in (path/snapshot).iterdir():
    if file.is_file():put(file,'reproduction/attempts/'+name+'/'+snapshot+'/'+file.name+'.txt',True)
for name in primary+diag:
 data=json.loads((r/name/'result.json').read_text());assert data['outcome']=='passed';assert all(data['guards'].values());assert data['fixture']['activeDayCards']==121
 assert next(x for x in data['initialArchive']['entries'] if x['name']=='master.md')['characters']==13000

 else:assert len(data['concurrent']['offers'])==120 and data['caughtAt']-data['startedAt']<600000
 field='diagnostic' if name in diag else 'primary';audit[field][name]={'outcome':data['outcome'],'guards':data['guards'],'finalExport':data['finalExport'],'initialArchive':data['initialArchive']}
expected=json.loads((r/'primary-probe-sha256.json').read_text())
for name in primary+diag:
 for file,value in expected.items():
  if file=='inspect-fixture.mjs':continue # independently executed validation tool, not the journey
  assert sha((r/name/'probe'/file).read_bytes())==value,(name,file)
secrets=[]
for folder in ['caughtup-seed','caughtup-seed-v2','caughtup-seed-v3','backlog-seed-v2','backlog-seed-v3']:
 p=r/folder/'fixture.json'
 if not p.exists():continue
 data=json.loads(p.read_text());audit['privateSeedIdentities'][folder]={'fixtureSha256':sha(p.read_bytes()),'aPosition':data['a'].get('position'),'bPosition':data['b'].get('position')}
 for device in ['a','b']:
  secrets.append(data[device]['code'].encode())
  for c in data[device]['credentials']:
   for key in ['privateKey','credentialId','userHandle']:
    if c.get(key):secrets.append(c[key].encode())

for file in dest.rglob('*'):
 if not file.is_file():continue
 content=file.read_bytes();content=gzip.decompress(content) if file.name.endswith('.gz') else content
 assert not any(s in content for s in secrets),('Private fixture value in evidence',str(file.relative_to(dest)))

(dest/'audit.json').write_text(json.dumps(audit,indent=2)+'\n');(dest/'artifact-provenance.json').write_text(json.dumps(manifest,indent=2)+'\n')
print(json.dumps({'files':sum(p.is_file() for p in dest.rglob('*')),'bytes':sum(p.stat().st_size for p in dest.rglob('*') if p.is_file()),'audit':audit['secretScan'],'primary':list(audit['primary']),'diagnostic':list(audit['diagnostic'])},indent=2))
