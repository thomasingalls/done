import test from 'node:test';import assert from 'node:assert/strict';import{mkdtempSync,readFileSync,rmSync}from'node:fs';import{tmpdir}from'node:os';import{join}from'node:path';import{preserveResult}from'./source/test/account-continuation-lab/result.mjs';
test('successful cleanup persists the original result exactly',async t=>{const dir=mkdtempSync(join(tmpdir(),'done-result-'));t.after(()=>rmSync(dir,{recursive:true,force:true}));const result={outcome:'passed',guards:{writing:true}};await preserveResult(join(dir,'result.json'),result,async()=>{});assert.deepEqual(JSON.parse(readFileSync(join(dir,'result.json'))),result)});
test('cleanup failure cannot mask the original failure or prevent durable evidence',async t=>{const dir=mkdtempSync(join(tmpdir(),'done-result-'));t.after(()=>rmSync(dir,{recursive:true,force:true}));const result={outcome:'failed',error:'original failure',traces:[{label:'original deadline',dataComplete:false}]},cause=Error('cleanup disconnected');await assert.rejects(preserveResult(join(dir,'result.json'),result,async()=>{throw cause}),e=>e===cause);const saved=JSON.parse(readFileSync(join(dir,'result.json')));assert.equal(saved.error,'original failure');assert.equal(saved.cleanupError,String(cause));assert.equal(saved.outcome,'failed');assert.deepEqual(saved.traces,result.traces)});
test('a dead control child rejects without scheduling a timeout or writing',async()=>{const text=readFileSync(new URL('./source/test/account-continuation-lab/runtime.mjs',import.meta.url),'utf8'),a=text.indexOf('export const control='),b=text.indexOf('\nexport async function start',a),source=text.slice(a,b).replace('export const control=','');const control=Function('child','setTimeout','serial','pending','return ('+source.replace(/;\s*$/,'')+');')({exitCode:1,stdin:{destroyed:true}},()=>{throw Error('must not schedule')},0,new Map());await assert.rejects(control('stop'),/Control unavailable stop/)});
