from pathlib import Path
import hashlib, io, json, subprocess, tarfile
root=Path(__file__).parent
repo=Path('/Users/thomasingalls/.codex/worktrees/7fda/done')
sha=lambda b:hashlib.sha256(b).hexdigest()
base='27dc7d78d973f2741a813d8d217f28e0c2f033a4'
paths=json.loads((repo/'docs/performance-experiments/2026-10-21-live-sync-attribution/source.json').read_text())['archivePaths']
archive=subprocess.check_output(['git','archive',base,*paths],cwd=repo)
original={}
with tarfile.open(fileobj=io.BytesIO(archive)) as t:
 for member in t:
  if member.isfile(): original[member.name]=sha(t.extractfile(member).read())
for file,h in original.items():assert sha((root/'baseline'/file).read_bytes())==h,file
versions={}
for version in ['baseline','candidate-v1','candidate-v2','candidate-v3','candidate-final','candidate-v4']:
 files=dict(original)
 for file in original: files[file]=sha((root/version/file).read_bytes())
 for file in ['src/crdt/receivePageTransport.ts']:
  if (root/version/file).exists():files[file]=sha((root/version/file).read_bytes())
 versions[version]=dict(sourceSha256=files,changes=[file for file,h in files.items() if original.get(file)!=h])
assert set(versions['candidate-v4']['changes'])=={'src/crdt/accountReceive.ts','src/crdt/receivePageTransport.ts','src/account/privateDeliveryHttp.ts'}
for folder in ['account-service','account-protocol']:
 assert all(versions['candidate-v4']['sourceSha256'][f]==h for f,h in original.items() if f.startswith(folder+'/'))
for file in versions['candidate-v4']['changes']:assert sha((repo/file).read_bytes())==versions['candidate-v4']['sourceSha256'][file]
probe={p.name:sha(p.read_bytes()) for p in (root/'baseline/test/live-sync-attribution').iterdir() if p.is_file()}
for name,h in probe.items():assert sha((root/'candidate-v4/test/live-sync-attribution'/name).read_bytes())==h
builds={}
for name in ['baseline-dist','candidate-v1-dist','candidate-v2-dist','candidate-v3-dist','candidate-final-dist','candidate-v4-dist']:
 builds[name]={str(p.relative_to(root/name)):sha(p.read_bytes()) for p in sorted((root/name).rglob('*')) if p.is_file()}
previous=json.loads((repo/'docs/performance-experiments/2026-10-21-live-sync-attribution/identity.json').read_text())
regressions={f:sha((repo/f).read_bytes()) for f in ['test/receivePageTransport.test.ts','test/account-receive/batch-transport.test.mjs','test/account-receive/groups-transport.test.mjs','test/account-receive-transport/fixtures.mjs','test/account-receive-transport/controls.test.mjs']}
result=dict(baseCommit=base,precedingPatch=json.loads((root/'starting.json').read_text()),environment=previous['environment'],browser=previous['browser'],archivePaths=paths,signedBaselineFiles=len(original),sources=versions,builds=builds,probeSha256=probe,regressionSha256=regressions,measuredProductionMatchesCheckout=True,serviceAndProtocolSourcesUnchanged=True,buildEnvironment=dict(APP_BUILD_REVISION=base,APP_BUILD_DATE='2026-10-17',VITE_ACCOUNT_BATCH='coordinated-2000'),notes='Build revision label is controlled metadata, not a signed candidate identity; exact candidate source/build hashes identify measured bytes. Lock agreement does not audit the whole installed package store.')
(root/'identity.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(dict(signedBaselineFiles=len(original),candidateChanges=versions['candidate-v4']['changes'],probeFiles=len(probe))))
