function watch(p:Page,label:string,rows:any[],tasks:Promise<any>[]){const pending=new Map();p.on('request',q=>{const path=new URL(q.url()).pathname;if(!path.startsWith('/account/'))return;const row:any={id:rows.length,label,path,phase,at:Date.now(),bodySha256:q.postDataBuffer()?digest(q.postDataBuffer()!):null};if(path.endsWith('/read-v2-batch')){const parts=decodeInboundBatch(new Uint8Array(q.postDataBuffer()!));row.batch=parts.map(bytes=>{const d=decodeInboundRequest(bytes);bytes.fill(0);return{after:d.after,limit:d.limit,self:d.reader===d.publisher,publisherSha256:digest(d.publisher)}});}else if(path.endsWith('/read-v2')){const d=decodeInboundRequest(new Uint8Array(q.postDataBuffer()!));Object.assign(row,{after:d.after,limit:d.limit,self:d.reader===d.publisher,publisherSha256:digest(d.publisher)});}rows.push(row);pending.set(q,row);});p.on('response',res=>{const row=pending.get(res.request());if(!row)return;row.headersAt=Date.now();row.status=res.status();if(!row.path.endsWith('/subscribe'))tasks.push((async()=>{try{const bytes=new Uint8Array(await res.body());row.bodyAt=Date.now();if(res.status()>=400){try{const v=JSON.parse(Buffer.from(bytes).toString());for(const k of ['kind','reason','refusal'])if(typeof v[k]==='string'&&/^[a-z-]{1,64}$/.test(v[k]))row[k]=v[k]}catch{row.unclassified=true;}}else if(row.path.endsWith('/submit')){const d=decodePositionResponse(bytes);row.kind=d.kind;if(d.kind==='accepted')row.position=d.head.position;}else if(row.path.endsWith('/read-v2-batch'))row.pages=decodeInboundBatchPages(bytes).map(b=>{const d=decodeInboundPage(b);b.fill(0);return{entries:d.entries.length,next:d.next,through:d.through}});else if(row.path.endsWith('/read-v2')){const d=decodeInboundPage(bytes);Object.assign(row,{entries:d.entries.length,next:d.next,through:d.through})}bytes.fill(0);}catch{row.bodyUnavailable=true}})());});p.on('requestfailed',q=>{const row=pending.get(q);if(row)row.failedAt=Date.now();});}
