function clean(v,remove){
 if(v===null||typeof v!=='object')return v;
 if(Array.isArray(v))return v.map(x=>clean(x,remove)).filter(x=>x!==undefined);
 if(remove&&v.type==='ImportDeclaration'&&v.source.value==='/experiment/observer.mjs')return undefined;
 if(remove&&v.type==='ExpressionStatement'&&v.expression.type==='CallExpression'&&v.expression.callee.type==='Identifier'&&v.expression.callee.name==='observe')return undefined;
 if(remove&&v.type==='AwaitExpression'&&v.argument.type==='CallExpression'&&v.argument.callee.type==='Identifier'&&v.argument.callee.name==='span'){
  assert.equal(v.argument.arguments.length,2);assert.equal(v.argument.arguments[0].type,'StringLiteral');assert.equal(v.argument.arguments[1].type,'ArrowFunctionExpression');assert.equal(v.argument.arguments[1].params.length,0);assert.notEqual(v.argument.arguments[1].body.type,'BlockStatement');return clean({...v,argument:v.argument.arguments[1].body},remove);
 }
 return Object.fromEntries(Object.entries(v).filter(([k])=>!excluded.has(k)).map(([k,x])=>[k,clean(x,remove)]));
}

}
