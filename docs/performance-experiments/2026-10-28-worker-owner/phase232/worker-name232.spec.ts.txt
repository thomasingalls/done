import {test,expect,type Page,type TestInfo} from '@playwright/test';
import {createdAccount} from '../device-enrollment/helpers';
import {openMenuSection} from '../geometry/openApp';
import {add} from './devices';
import {startService,stopService} from './server';
async function nameFlow(page:Page,info:TestInfo,write:boolean){
 page.setDefaultTimeout(15000);page.on('console',m=>{if(m.text().startsWith('D232:'))console.log(m.text());});
 const service=await startService(info.outputPath('service'),undefined,true,true);
 const observations:Record<string,unknown>={write};
 try{
  const first=await createdAccount(page);
  if(write)await add(page,'Name-test immediate writing');
  observations['currentUser']=await page.locator('.menu-metadata .menu-row .menu-value').first().textContent();
  observations['beforeClose']=await page.getByLabel('User name',{exact:true}).inputValue();
  await expect(page.getByLabel('User name',{exact:true})).toHaveValue('Device owner');
  await openMenuSection(page,'Account');
  await page.getByRole('button',{name:'Use another account',exact:true}).click();
  await expect(page.getByRole('heading',{name:'Choose an account',exact:true})).toBeVisible();
  observations['chooser']=await page.getByRole('button',{name:/^Unlock /}).allTextContents();
  await expect(page.getByRole('button',{name:/^Unlock Device owner/})).toBeVisible();
  await page.getByRole('button',{name:/^Unlock Device owner/}).click();
  await page.getByLabel('Unlock code',{exact:true}).fill(first.code);
  await page.getByRole('button',{name:'Confirm code and unlock with passkey',exact:true}).click();
  await expect(page.locator('.shell')).toBeVisible();
  observations['afterReopen']=await page.getByLabel('User name',{exact:true}).inputValue();
  await expect(page.getByLabel('User name',{exact:true})).toHaveValue('Device owner');
  if(write)await expect(page.locator('.shell')).toContainText('Name-test immediate writing');
 }finally{
  observations['afterCase']=await page.evaluate(()=>{const dom={draft:(document.querySelector('[aria-label="User name"]') as HTMLInputElement)?.value,user:document.querySelector('.menu-metadata .menu-row .menu-value')?.textContent};const e=document.querySelector('.shell');if(!e)return{dom,noShell:true};const k=Object.keys(e).find(k=>k.startsWith('__reactFiber$'));if(!k)return{dom,noFiber:true};let f=(e as any)[k];for(let i=0;f&&i<50;i++,f=f.return){const p=f.memoizedProps,v=p?.workspace?p:p?.value,w=v?.workspace;if(w&&typeof w.read==='function'&&w.status)return{dom,names:w.read().actors.map((a:any)=>a.name),snapshotNames:v.snapshot?.actors.map((a:any)=>a.name),status:w.status};}return{dom,noWorkspace:true};}).catch(()=>({closed:true}));
  console.log('AFTER232',JSON.stringify(observations));
  await info.attach('public-name-observations',{body:JSON.stringify(observations),contentType:'application/json'});
  await stopService(service);
 }
}
test('native setup retains the entered user name through immediate empty close and reopen',async({page},info)=>{test.setTimeout(60000);await nameFlow(page,info,false);});
test('native setup retains the entered user name after immediate first writing, close and reopen',async({page},info)=>{test.setTimeout(60000);await nameFlow(page,info,true);});
