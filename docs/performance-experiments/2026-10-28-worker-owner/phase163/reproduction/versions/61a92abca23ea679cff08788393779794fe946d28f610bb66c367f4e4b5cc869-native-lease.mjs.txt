async function hold(locks,name,mode,signal){
 const ready=deferred(),release=deferred();let active=false,flight;

  const stop=()=>{active=false;release.resolve()};signal?.addEventListener('abort',stop,{once:true});active=true;
  const lease={get state(){return active?'active':'closed'},write(effect){if(!active)throw Error('inactive-native-lease');return effect()},close(){stop();return flight}};
  ready.resolve(lease);if(signal?.aborted)stop();try{await release.promise}finally{active=false;signal?.removeEventListener('abort',stop)};
 });

}
/** Lab-only topology: reservation serializes new supervisors; the exclusive old-name barrier drains old/orphan workers. */
export async function acquireBridgeLease(locks,slotId,signal){

}
