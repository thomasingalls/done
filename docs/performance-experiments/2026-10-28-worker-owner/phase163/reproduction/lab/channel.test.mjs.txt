import {test}from'node:test';import assert from'node:assert/strict';
import{createChannel,parseEnvelope,MAX_MESSAGE_BYTES,MAX_PENDING}from'./channel.mjs';
const owner='a'.repeat(32),other='b'.repeat(32),reply=(r,kind='result')=>({...r,kind});
const ready=()=>{const g=createChannel(owner),q=g.issue('open');assert.equal(g.receive(reply(q)).kind,'accepted');return g};
test('fresh bounded opening and command only after ready',()=>{const g=createChannel(owner);assert.throws(()=>g.issue('snapshot'));const q=g.issue('open');assert.equal(g.phase,'opening');assert.throws(()=>g.issue('open'));assert.equal(g.receive(reply(q)).kind,'accepted');assert.equal(g.phase,'ready');});
test('different incarnation cannot fulfill pending request',()=>{const g=ready(),q=g.issue('snapshot');assert.equal(g.receive({...reply(q),owner:other}).kind,'stale');assert.equal(g.pending,1);assert.equal(g.receive(reply(q)).kind,'accepted');});
test('duplicate and wrong operation replies never satisfy a different request',()=>{const g=ready(),q=g.issue('snapshot');assert.equal(g.receive({...reply(q),operation:'probe'}).kind,'stale');assert.equal(g.receive(reply(q)).kind,'accepted');assert.equal(g.receive(reply(q)).kind,'stale');});
test('close immediately invalidates every earlier reply and requires exact closed acknowledgment',()=>{const g=ready(),q=g.issue('append'),close=g.issue('close');assert.equal(g.pending,1);assert.equal(g.receive(reply(q)).kind,'stale');assert.throws(()=>g.issue('snapshot'));assert.equal(g.receive(reply(close,'closed')).kind,'accepted');assert.equal(g.phase,'closed');});
test('close can cancel an opening and late ready cannot resurrect it',()=>{const g=createChannel(owner),q=g.issue('open'),c=g.issue('close');assert.equal(g.receive(reply(q)).kind,'stale');assert.equal(g.receive(reply(c,'closed')).kind,'accepted');});
test('failure closes admission until terminal close or a new incarnation',()=>{const g=ready(),q=g.issue('probe');g.receive(reply(q,'refused'));assert.equal(g.phase,'failed');assert.throws(()=>g.issue('append'));assert.equal(g.receive(reply(g.issue('close'),'closed')).kind,'accepted');});
test('pending queue capacity is bounded and close is always available',()=>{const g=ready();for(let i=0;i<MAX_PENDING;i++)g.issue('snapshot');assert.throws(()=>g.issue('probe'),/capacity/);assert.doesNotThrow(()=>g.issue('close'));});
test('commands copy caller-owned payload before publication',()=>{const g=ready(),b=new Uint8Array([1]),q=g.issue('append',b);b[0]=2;assert.equal(q.bytes[0],1)});
test('oversize and shared payloads refuse without consuming sequence or state',()=>{const g=createChannel(owner);assert.throws(()=>g.issue('open',new Uint8Array(MAX_MESSAGE_BYTES+1)));assert.equal(g.phase,'new');assert.equal(g.issue('open').id,1);assert.throws(()=>parseEnvelope({version:1,owner,id:2,operation:'probe',bytes:new Uint8Array(new SharedArrayBuffer(4))}));});
test('closed schema rejects extra fields, inherited-only fields and accessors without invoking them',()=>{const q={version:1,owner,id:1,operation:'open',bytes:new Uint8Array()};assert.throws(()=>parseEnvelope({...q,secret:'x'}));assert.throws(()=>parseEnvelope(Object.create(q)));let called=false;const x={...q};Object.defineProperty(x,'owner',{get(){called=true;throw Error()}});assert.throws(()=>parseEnvelope(x));assert.equal(called,false)});
test('invalid sequence, version, method and owner are refused',()=>{for(const change of[{id:0},{id:1.1},{version:2},{operation:'eval'},{owner:'../x'}])assert.throws(()=>parseEnvelope({version:1,owner,id:1,operation:'open',bytes:new Uint8Array(),...change}));});
test('detached buffers are refused',()=>{const b=new Uint8Array([1]);structuredClone(b,{transfer:[b.buffer]});assert.throws(()=>parseEnvelope({version:1,owner,id:1,operation:'open',bytes:b}));});
test('terminal invalidation never accepts queued success',()=>{const g=ready(),q=g.issue('append');g.invalidate();assert.equal(g.receive(reply(q)).kind,'stale');assert.throws(()=>g.issue('open'));});

test('terminal reply cannot close an unrelated operation',()=>{const g=ready(),q=g.issue('snapshot');assert.equal(g.receive(reply(q,'closed')).kind,'invalid');assert.equal(g.phase,'ready');assert.equal(g.pending,1)});
