import{createDeviceEnrollment}from'./src/account/deviceEnrollment';
import{createMembershipEnrollment}from'./src/account/membershipEnrollment';
import{createAccountEnrollment}from'./src/account/accountEnrollment';
import{hasJoinedAccount}from'./src/crdt/joinedAccount';
export function registerEnrollmentPort(){
 window.__enrollmentPortLab={async open(kind){const spec=window.spec,joined=await hasJoinedAccount(spec.slotId,spec.code);if(!joined)throw Error('joined fixture required');const config={pin:spec.pin,networkSessions:true,membershipContinuation:true};const owner=kind==='selector'?await createAccountEnrollment(config,spec.slotId,spec.code,'new',joined.workspaceId):kind==='membership'?createMembershipEnrollment(config,spec.slotId,spec.code,'new',joined.workspaceId):createDeviceEnrollment(config,spec.slotId,spec.code,'new',joined.workspaceId);window.__enrollmentPortOwner=owner;const view=await owner.refresh();if(view.kind!=='editing'||!view.editor)throw Error('original joined editor not ready: '+view.kind);window.__enrollmentPortWorkspace=view.editor.workspace;return{archive:view.editor.workspace.exportEntries(),actorId:view.editor.actorId,accountId:view.editor.accountId}},async input(text){const workspace=window.__enrollmentPortWorkspace,field=workspace.text({kind:'master'}),n=field.read().text.length;await field.input({name:'insert',text},{ranges:[{anchor:n,head:n}],main:0}).accepted;await workspace.whenDurable();return workspace.exportEntries()},close:()=>window.__enrollmentPortOwner.close()};
}
import{createWorkerAccountEnrollment}from'./src/ui/accountEnrollmentWorker';
window.__enrollmentWorkerLab={async open(){const spec=window.spec,joined=await hasJoinedAccount(spec.slotId,spec.code);if(!joined)throw Error('joined fixture required');const owner=await createWorkerAccountEnrollment({pin:spec.pin,networkSessions:true,membershipContinuation:true},spec.slotId,spec.code,{role:'new',workspaceId:joined.workspaceId},window.__testStop.signal);window.__enrollmentPortOwner=owner;const view=await owner.refresh();if(view.kind!=='editing'||!view.editor)throw Error('native joined editor not ready: '+view.kind);window.__enrollmentPortWorkspace=view.editor.workspace;return{archive:view.editor.workspace.exportEntries(),actorId:view.editor.actorId,accountId:view.editor.accountId}},input:(...args)=>window.__enrollmentPortLab.input(...args),close:()=>window.__enrollmentPortLab.close()};
