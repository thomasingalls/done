export function bounded(value){let bytes=0,nodes=0;const seen=new Set(),todo=[value];while(todo.length){if(++nodes>100000)return false;const v=todo.pop();if(v==null||typeof v==='boolean'||typeof v==='number')continue;if(typeof v==='string'){bytes+=v.length*2}else if(v instanceof Uint8Array&&v.buffer instanceof ArrayBuffer){try{v.subarray(0,0)}catch{return false}bytes+=v.byteLength}else if(typeof v==='object'){if(seen.has(v))return false;seen.add(v);if(!Array.isArray(v)&&Object.getPrototypeOf(v)!==Object.prototype)return false;for(const d of Object.values(Object.getOwnPropertyDescriptors(v))){if(!('value'in d))return false;todo.push(d.value)}}else return false;if(bytes>4*1024*1024)return false;}return true;}
