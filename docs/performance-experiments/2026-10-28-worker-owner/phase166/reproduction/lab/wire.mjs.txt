export function inspectWire(value){let bytes=0,nodes=0;const seen=new Set(),todo=[value];const no=reason=>({ok:false,reason,bytes,nodes});while(todo.length){if(++nodes>100000)return no('nodes');const v=todo.pop();if(v==null||typeof v==='boolean'||typeof v==='number')continue;if(typeof v==='string'){bytes+=v.length*2}else if(v instanceof Uint8Array&&v.buffer instanceof ArrayBuffer){try{v.subarray(0,0)}catch{return no('detached')}bytes+=v.byteLength}else if(typeof v==='object'){if(seen.has(v))return no('repeated-object');seen.add(v);if(!Array.isArray(v)&&Object.getPrototypeOf(v)!==Object.prototype)return no('prototype');for(const d of Object.values(Object.getOwnPropertyDescriptors(v))){if(!('value'in d))return no('accessor');todo.push(d.value)}}else return no('type');if(bytes>4*1024*1024)return no('bytes');}return{ok:true,bytes,nodes};}
export const bounded=value=>inspectWire(value).ok;
