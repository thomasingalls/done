import assert from 'node:assert/strict';

writeFileSync(root+'/docs/performance-experiments/2026-10-28-resumable-preparation/phase127/crypto-review.json',JSON.stringify({baseTree:base,previousSources:a.size,currentSources:b.size,addedSources:[...b.keys()].filter(p=>!a.has(p)),previousDomains:before.domains.length,currentDomains:next.domains.length,allPriorSourcesAndConstructionRecordsUnchanged:true,newOpenConstruction:'account-local-preparation'},null,2)+'\n');
