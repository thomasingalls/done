import{test,expect,chromium}from'@playwright/test';import{cpSync,readFileSync,writeFileSync}from'node:fs';import{join}from'node:path';import{root,card,exported,archiveIdentity}from'./common';import{start,stop}from'./runtime.mjs';import{restoreAuthenticator,openSaved}from'./devices';
test('read failed profiles only through offline detached copies',async()=>{const out=process.env['FAIR_OUT']!,source=root+'/maintenance-3-C-withheld',fixture=JSON.parse(readFileSync(root+'/prepared-fixture/fixture.json','utf8')),account=fixture.accounts.find((a:any)=>a.label==='H');const rows:any[]=[];const state=join(out,'state');cpSync(source+'/state',state,{recursive:true,errorOnExist:true});try{await start(state,root+'/source-C/dist');for(const index of[2,1]){const profile=join(out,'profile-H'+index);cpSync(source+'/private-profile-H'+index,profile,{recursive:true,errorOnExist:true});const ctx=await chromium.launchPersistentContext(profile,{executablePath:process.env['CHROMIUM_PATH']!,offline:true,baseURL:'http://localhost:53781',viewport:{width:1440,height:1000}});try{const page=ctx.pages()[0]??await ctx.newPage();const saved=account.actors[index];await restoreAuthenticator(page,saved.credentials);await page.goto('/');await openSaved(page,saved.code,true);await page.getByLabel('Go to date',{exact:true}).fill(account.date);const text=await card(page,'H',true).innerText();const archive=await exported(page);const row={index,followupInCard:text.includes(' Undo safety guard;'),followupInExport:Object.values(archive).some(v=>v.includes(' Undo safety guard;')),archive:archiveIdentity(archive),status:await page.evaluate(()=>{const x=(window as any).__maintenanceOwner;return{workspaceKeys:Object.keys(x.workspace.status??{}),workspaceDurable:x.workspace.status?.durable,sync:x.sync?.status};})};rows.push(row);}finally{await ctx.close();}}writeFileSync(join(out,'diagnostic.json'),JSON.stringify({noStorageReinjection:true,offline:true,originalProfilesUntouched:true,rows},null,2)+'\n');}finally{await stop();}});
