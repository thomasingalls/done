import{test,expect,chromium}from'@playwright/test';import{writeFileSync}from'node:fs';import{join}from'node:path';import{installDispatchProbe,browserIdentity,clockBrackets}from'./dispatch-probe';
test('isolated native input and known blocked-input sensitivity',async()=>{
 const out=process.env['FAIR_OUT']!,fault=process.env['DISPATCH_FAULT']==='1',result:any={outcome:'starting',target:'plain',fault,offers:[],clocks:[]};let browser:any;
 try{browser=await chromium.launch({executablePath:process.env['CHROMIUM_PATH']!});const ctx=await browser.newContext({viewport:{width:1440,height:1000}});result.root=await browserIdentity(browser);await ctx.addInitScript(installDispatchProbe,{fault});const p=await ctx.newPage();await p.goto('data:text/html,<title>Disposable dispatch control</title><body></body>');await p.bringToFront();await p.evaluate(()=>(window as any).__dispatch.plain());result.clocks.push(await clockBrackets(p));result.startedAt=Date.now();
 for(let i=0;i<120;i++){const scheduledAt=result.startedAt+i*(fault?150:500);await p.waitForTimeout(Math.max(0,scheduledAt-Date.now()));const row:any={kind:'key',index:i,scheduledAt,at:Date.now()};await p.keyboard.press('w');row.returnedAt=Date.now();result.offers.push(row);}result.typingFinishedAt=Date.now();await p.evaluate(()=>new Promise(resolve=>requestAnimationFrame(()=>requestAnimationFrame(resolve))));expect(await p.locator('#dispatch-control').inputValue()).toBe('w'.repeat(120));result.dispatch=await p.evaluate(()=>(window as any).__dispatch.read());result.clocks.push(await clockBrackets(p));result.driver=(globalThis as any).__dispatchNodeSnapshot?.();expect(result.dispatch.events.filter((e:any)=>e.kind==='beforeinput')).toHaveLength(120);expect(result.dispatch.plainRows).toHaveLength(120);expect(result.dispatch.events.every((e:any)=>e.trusted&&e.focused&&e.visible==='visible'&&e.active==='plain')).toBe(true);if(fault)expect(result.dispatch.faults.map((e:any)=>e.index)).toEqual([30,60,90]);result.outcome='passed';
 }catch(error){result.outcome='failed';result.error=String(error);throw error;}finally{await browser?.close();writeFileSync(join(out,'result.json'),JSON.stringify(result,null,2)+'\n');}
});
