import{readFileSync,writeFileSync}from'node:fs';import assert from'node:assert/strict';
import{inputLatencies,submissionCadence,displayedBacklog}from'./source/scripts/lib/sync-cadence-analysis.mjs';
const root=new URL('./',import.meta.url),read=p=>JSON.parse(readFileSync(new URL(p,root))),rows=[];
for(const item of read('run-plan.json')){
 const x=read(item.name+'/result.json');assert.equal(x.outcome,'passed');assert.equal(x.variant,item.variant);assert.equal(x.diagnosticLoadBoundaryMs,90000);assert.equal(x.backgroundReplayPassed,null);assert.equal(x.backgroundReplayCensored,true);assert(x.loadStoppedAt>=x.startedAt+90000);assert(x.loadStoppedAt-x.startedAt<110000);assert(Object.values(x.guards).every(Boolean));assert(x.offlineColdWriterReopen.sameProfile&&x.offlineColdWriterReopen.noStorageStateReinjection&&x.offlineColdWriterReopen.retainedDeliveryByteExact);assert.deepEqual(x.pendingBeforeClose.archive,x.offlineColdWriterReopen.archive);assert(x.automaticOriginalWritingRecovered&&x.originalWholeExportEquality&&x.automaticPendingConverged);assert(x.originalRecoveredAt-x.loadStoppedAt<=120000&&x.pendingConvergedAt-x.pendingReconnectAt<=120000);
 const inputs=inputLatencies(x);const foregroundAt=x.foregroundObservedAt;const backlog=displayedBacklog(x.probesAtDeadline[0].progress,x.probesAtDeadline[1].progress,x.startedAt,foregroundAt);const native=x.traces.flatMap(t=>t.events);assert(native.every(e=>e.tag==='http-result'));rows.push({...item,inputs,cadence:submissionCadence(x.network),backlog,foregroundDelivery:x.originalDeliveryPassed,backgroundReplay:'censored at fixed90s; no600s acceptance inference',catchups:x.catchups,quiet429:x.network.filter(v=>v.label.startsWith('Q')&&v.status===429).length,guards:x.guards,clocks:x.clocks,metrics:x.metrics,loadDurationMs:x.loadStoppedAt-x.startedAt});
}
const conditions=Object.fromEntries('ABCD'.split('').map(label=>[label,rows.filter(v=>v.variant===label).map(v=>({name:v.name,peerComplete:v.inputs.peer.complete,peerP95Ms:v.inputs.peer.complete?v.inputs.peer.observed.p95:null,localP95Ms:v.inputs.local.observed.p95,requests:v.cadence.requests,uniqueBodies:v.cadence.uniqueBodies,submitP95Ms:v.cadence.observedLatencyMs.p95,displayedMaxBacklog:v.backlog.maxPendingDelta,displayedBacklogAreaPositionMs:v.backlog.backlogAreaPositionMs}))]));
writeFileSync(new URL('analysis.json',root),JSON.stringify({status:'passed diagnostic evidence, no promotion',conditions,runs:rows,decision:'Original23 held; finite2x2 diagnostic only, full600s gates not measured.'},null,2)+'\n');console.log(JSON.stringify(conditions,null,2));
