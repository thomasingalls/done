async function census(db:IDBDatabase){
 const data=await new Promise<any>((yes,no)=>{const tx=db.transaction(['meta','content','heads','outbound']);const rows:any={};for(const name of ['meta','content','heads','outbound']){const s=tx.objectStore(name),k=s.getAllKeys(),v=s.getAll();v.onsuccess=()=>rows[name]=k.result.map((key,i)=>[key,v.result[i]])}tx.oncomplete=()=>yes(rows);tx.onabort=()=>no(tx.error)});
