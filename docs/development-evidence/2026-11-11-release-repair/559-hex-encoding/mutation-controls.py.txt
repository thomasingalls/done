#!/usr/bin/env python3
"""Phase 559 mutation controls. Each mutation edits one site of the final source text (the two hex
routines and the legacy source observation), runs the
whole of test/hexEncoding.test.ts, lists the cases that failed, and restores the file byte for byte.
Usage: python3 -I mutation-controls.py <repo root>  (prints one block per mutation)"""
import hashlib
import subprocess
import sys
from pathlib import Path

root = Path(sys.argv[1])
TEST = 'test/hexEncoding.test.ts'
CUSTODY = 'src/crdt/custodyCrypto.ts'
PROTOCOL = 'account-protocol/recovery.ts'
LEGACY = 'src/crdt/legacySource.ts'

MUTATIONS = [
    # Each path, and the choice between them.
    ('custodyCrypto: the join path for every length (the base routine)', CUSTODY,
     'if (b.length < HEX_SLOTS_FROM) {', 'if (b.length > 0) {'),
    ('recovery: the join path for every length (the base routine)', PROTOCOL,
     'if (value.length < HEX_SLOTS_FROM) {', 'if (value.length >= 0) {'),
    ('custodyCrypto: the slot path for every length (phase 559 as first written)', CUSTODY,
     'const HEX_SLOTS_FROM = 64;', 'const HEX_SLOTS_FROM = 0;'),
    ('recovery: the slot path for every length (phase 559 as first written)', PROTOCOL,

    ('custodyCrypto: the threshold comparison inverted', CUSTODY,
     'if (b.length < HEX_SLOTS_FROM) {', 'if (b.length >= HEX_SLOTS_FROM) {'),
    ('recovery: the threshold comparison inverted', PROTOCOL,

     "const HEX_BYTES = Array.from({ length: 256 }, (_, byte) => byte.toString(16).padStart(2, '0'));",

     "ascii[2 * n + 1] = '0123456789abcdef'.charCodeAt(n & 15);",

def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


results = []
for name, rel, old, new in MUTATIONS:
    path = root / rel
    original = path.read_bytes()
    text = original.decode()
    if text.count(old) != 1:
        print(f'== {name}: anchor found {text.count(old)} times; not run')
        results.append((name, 'not run'))
        continue
    before = sha(path)
    try:
        path.write_text(text.replace(old, new))
        run = subprocess.run(
            ['timeout', '300', 'node_modules/.bin/vitest', 'run', TEST, '--reporter=verbose'],
            cwd=root, capture_output=True, text=True)
        out = run.stdout + run.stderr
        failed = [l.strip() for l in out.splitlines() if l.strip().startswith('×')]
        passed = [l.strip() for l in out.splitlines() if l.strip().startswith('✓')]
        print(f'== {name} ({rel}); vitest exit {run.returncode}; {len(failed)} failed, {len(passed)} passed')
        for l in failed:
            print('  FAILED ' + ' > '.join(l[2:].split(' > ')[-2:]))
        results.append((name, f'tripped ({len(failed)} cases)' if failed else 'NOT TRIPPED'))
    finally:
        path.write_bytes(original)
        assert sha(path) == before, f'{rel} not restored'
print('== summary')
for name, outcome in results:
    print(f'  {outcome}: {name}')
print('all sources restored byte for byte')
