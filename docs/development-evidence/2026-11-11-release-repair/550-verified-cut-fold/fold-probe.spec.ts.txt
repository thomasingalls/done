// COPY-ONLY probe (fold after a verified install): a device ahead on a peer stream it has reduced and behind on a
// pruned stream. MODE=strand (tip without the fix: stays stranded until a newer snapshot),
// MODE=recover (fix: the cut is verified by reading the peer's chain, the snapshot installs and the
// device catches up at once), MODE=unverifiable (fix, with the peer's chain reads answered 409
// `unavailable`: a named notice appears, stays while writing keeps arriving, and clears when a
// newer snapshot installs). Copy-only seams in the build: `__holdHistory`, `__allowReduction`,
// retention thresholds 1 / 3 s / 1 s / 2 s, and the export-state hook used for convergence.

const MODE: string = process.env.MODE ?? 'compat';
const NAMED = 'could not be checked against the history this device kept';
const reads = (body: string) =>
  [
    ...body.matchAll(
      /\["done\/private-read\/v2",2,"[^"]*","[^"]*","[0-9a-f]+","([0-9a-f]{32})","([0-9a-f]{32})","[0-9a-f]+","[0-9a-f]+",\d+,(\d+),(\d+)/g,
    ),
  ].map((m) => ({ reader: m[1]!, publisher: m[2]!, after: Number(m[3]), limit: Number(m[4]) }));
const workers = async (p: Page, fn: () => void) => {
  for (const w of p.workers()) await w.evaluate(fn).catch(() => {});
};
async function baseline(p: Page) {
  return p.evaluate(async () => {
    const out: { name: string; v: number; revision: number }[] = [];
    for (const { name } of await indexedDB.databases()) {
      if (!name) continue;
      const db = await new Promise<IDBDatabase>((yes, no) => {
        const r = indexedDB.open(name);
        r.onsuccess = () => yes(r.result);
        r.onerror = () => no(r.error);
      });
      try {
        if (!db.objectStoreNames.contains('meta')) continue;
        const c = await new Promise<any>((yes, no) => {
          const tx = db.transaction('meta');
          const g = tx.objectStore('meta').get('counter');
          tx.oncomplete = () => yes(g.result);
          tx.onabort = () => no(tx.error);
        });
        if (c && typeof c.v === 'number')
          out.push({ name, v: c.v, revision: c.baseline?.revision ?? 0 });
      } finally {
        db.close();
      }
    }
    return out;
  });
}

      await expect(A.locator('.shell')).toContainText('Unsent writing on the third device', { timeout: 60000 });
      note(`submission accepted ${((Date.now() - released) / 1000).toFixed(1)} s after the network allowed it`);
      await expect.poll(sawA, { timeout: 90000, intervals: [200] }).toBe(true);
      const tc = Date.now();
      note(`installed ${((tc - released) / 1000).toFixed(1)} s after the submission could be sent`);
      const w = await waitFold(tc, reducedTo, 240000);
      note(`fold ${w} s after that installation`);
      expect(w).toBeDefined();
      expect(await olderOpens('tip')).toBe(true);
    }
    if (MODE === 'step' || MODE === 'hidden') {
      // Held: the snapshot is checked but not committed; the store stays openable by older builds.
      await Cd.waitForTimeout(35000);
      note(`${MODE}: committed? ${await sawA()}; revision ${await revision()}; notice ${JSON.stringify(await sentence())}`);
      expect(await sawA()).toBe(false);
      if (MODE === 'hidden') {
        await Cd.evaluate(() => {
          Object.defineProperty(document, 'visibilityState', { configurable: true, get: () => 'visible' });
          Object.defineProperty(document, 'hidden', { configurable: true, get: () => false });
          document.dispatchEvent(new Event('visibilitychange'));
        });
      } else {
        // Older builds open the held store.
        expect(await olderOpens('tip')).toBe(true);
        await workers(Cd, () => {
          (globalThis as any).__allowReduction = true;
        });
        // Finish the step: cancel the saved approval.
        await openMenuSection(Cd, 'Account');
        await Cd.getByRole('button', { name: 'Devices', exact: true }).click();
        await Cd.getByRole('button', { name: 'Connect another device', exact: true }).click();
        const unlockField = Cd.getByLabel('Unlock code', { exact: true });
        if (await unlockField.isVisible({ timeout: 5000 }).catch(() => false)) await unlock(Cd, next.code);
        await Cd.getByRole('button', { name: 'Cancel connection', exact: true }).click({ timeout: 30000 });
        note('third cancelled its saved approval');
        await Cd.getByRole('link', { name: 'Close and return to accounts' }).click().catch(() => {});
        await Cd.goto('/');
        await openSaved(Cd, next.code, true);
        await workers(Cd, () => {
          (globalThis as any).__allowReduction = true;
        });
      }
      const cleared = Date.now();
      await expect.poll(sawA, { timeout: 90000, intervals: [200] }).toBe(true);
      const tc = Date.now();
      note(`${MODE}: installed ${((tc - cleared) / 1000).toFixed(1)} s after the block cleared`);
      const w = await waitFold(tc, reducedTo);
      note(`${MODE}: fold ${w} s after that installation`);
      expect(w).toBeDefined();
      expect(await olderOpens('tip')).toBe(true);
    }
    if (MODE === 'capacity') {
      await Cd.waitForTimeout(40000);
      note(`capacity: revision ${await revision()}; panel ${JSON.stringify(await panelText())}; notice ${JSON.stringify(await sentence())}`);
      await leave();
      await serve('tip');
      note(`tip while the fold is parked for capacity opens? ${await open('tip')}`);
      await leave(false);
      await serve('patched');
      expect(await open('patched')).toBe(true);
      await workers(Cd, () => {
        (globalThis as any).__allowReduction = true;
      });
      await Cd.waitForTimeout(20000);
      note(`capacity after reopening: revision ${await revision()}`);
      // The ceiling lifts (seam off): the parked fold runs after new history and the retry clock.
      await workers(Cd, () => {
        (globalThis as any).__forceCapacity = false;
      });
      await add(B, 'B writing after the ceiling lifted');
      const lifted = Date.now();
      const w = await waitFold(lifted, reducedTo, 240000);
      note(`capacity: fold ${w} s after new history once the ceiling lifted`);
    }
    // Convergence: every device's text and exact Yjs state.
    await add(A, 'Final writing from A');
    for (const p of [B, Cd])
      await expect(p.locator('.shell')).toContainText('Final writing from A', { timeout: 60000 });
    await Cd.waitForTimeout(5000);
    const docs = [];
    for (const p of [A, B, Cd]) docs.push(await documentOf(p));
    note(
      'convergence: text ' +
        (docs[0]!.text === docs[1]!.text && docs[1]!.text === docs[2]!.text) +
        ', state ' +
        (JSON.stringify(docs[0]!.state) === JSON.stringify(docs[1]!.state) &&
          JSON.stringify(docs[1]!.state) === JSON.stringify(docs[2]!.state)),
    );
    expect(docs[2]!.text).toBe(docs[0]!.text);
    expect(docs[2]!.state).toEqual(docs[0]!.state);
  } finally {
    writeFileSync(
      info.outputPath('timeline.txt'),
      log.join('\n') + '\n' + 'errors: ' + JSON.stringify(errors) + '\n',
    );
    try {
      await thirdContext.close().catch(() => {});
      await pair?.other.close().catch(() => {});
    } finally {
      await stopService(service);
    }
  }
});
