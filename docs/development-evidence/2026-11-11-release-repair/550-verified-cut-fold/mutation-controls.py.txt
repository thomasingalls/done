for name, (f, a, b, test) in M.items():
    p = os.path.join(T, f); s = open(p).read()
    assert s.count(a) == 1, (name, a)
    open(p, 'w').write(s.replace(a, b))
    try:
        r = subprocess.run(['node', 'node_modules/vitest/vitest.mjs', 'run', test], cwd=T, capture_output=True, text=True, timeout=600)
        out = r.stdout + r.stderr
        failed = [l.strip() for l in out.splitlines() if l.strip().startswith('×')]
        summary = [l.strip() for l in out.splitlines() if l.strip().startswith('Tests ')]
        print(f'{name}: exit {r.returncode}; {summary[-1] if summary else "?"}')
        for l in failed: print('   ', l[:150])
    finally:
        open(p, 'w').write(s)
