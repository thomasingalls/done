+/** Test builds only (`VITE_ACCOUNT_TEST_HOOKS=1`): browser specs that must reach a state real use
+ * reaches slowly, such as a device reduced past a snapshot cut. A release build never sets the
+ * flag, so every hook reads as absent there. Hooks only withhold or allow scheduling; no check,
+ * proof or commit is skipped by them. */
+interface TestHooks {
+  /** Receive passes run without a history catalog check. */
+  holdHistory?: boolean;
+  /** Growth threshold for automatic local reduction, in place of 1,024 records. */
+  minRecords?: number;
+}
+const hooks = (): TestHooks | undefined =>
+  import.meta.env?.['VITE_ACCOUNT_TEST_HOOKS'] === '1'
+    ? (globalThis as { __doneTestHooks?: TestHooks }).__doneTestHooks
+    : undefined;
+export const testHoldsHistory = () => hooks()?.holdHistory === true;
+export const testMinRecords = () => {
+  const n = hooks()?.minRecords;
+  return typeof n === 'number' && Number.isSafeInteger(n) && n >= 1 ? n : undefined;
+};

+function storedZipText(buf: Buffer) {
+  const out: [string, string][] = [];
+  let at = 0;
+  while (at + 30 <= buf.length && buf.readUInt32LE(at) === 0x04034b50) {
+    if (buf.readUInt16LE(at + 8) !== 0) throw Error('compressed entry');
+    const size = buf.readUInt32LE(at + 18),
+      nameLength = buf.readUInt16LE(at + 26),
+      extra = buf.readUInt16LE(at + 28);
+    const name = buf.subarray(at + 30, at + 30 + nameLength).toString();
+    const start = at + 30 + nameLength + extra;
+    out.push([name, buf.subarray(start, start + size).toString()]);
+    at = start + size;
+  }
+  return out
+    .sort(([a], [b]) => a.localeCompare(b))
+    .map(([, text]) => text)
+    .join('\n');

+            () => true,
+            () => false,
+          );
+        note(`${name}: sends ${sent}, receives ${got}`);
+      }
+      await leave();
+      await serve('patched');
+      expect(await open('patched')).toBe(true);
+      return ok;
+    }
+    const panelText = () =>
+      Cd.getByRole('region', { name: 'History on this device' })
+        .allTextContents()
+        .then((t) => t.join(' '));
+    async function waitFold(since: number, from: number, timeout = 180000) {
+      const deadline = Date.now() + timeout;
+      while (Date.now() < deadline) {
+        if ((await revision().catch(() => from)) > from) return (Date.now() - since) / 1000;
+        await Cd.waitForTimeout(250);
+      }
+      return undefined;
+    }
+    // ---------------- reduce the third device past the snapshot cut ----------------
+    const reducedFrom = await revision();
+    await workers(Cd, () => {
+      (globalThis as any).__doneTestHooks = { holdHistory: true, minRecords: 1 };
+    });
+    await thirdContext.setOffline(true);
+    let reducedTo = reducedFrom;
+    for (let i = 0; i < 36 && reducedTo <= reducedFrom; i++) {
+      await Cd.waitForTimeout(5000);
+      reducedTo = await revision();
+    }
+    expect(reducedTo).toBeGreaterThan(reducedFrom);
+    // From here only a due fold can reduce: the growth threshold is back to 1,024 records and the
+    // 30-minute cooldown has just started.
+    await workers(Cd, () => {
+      (globalThis as any).__doneTestHooks = {};
+    });
+    note(`third reduced: revision ${reducedFrom} -> ${reducedTo}`);
+    blockA = false;
+    phase = 'after';
+    // The panel is opened before any measurement so no click counts as activity later.
+    await openMenuSection(Cd, 'Data');
+    Cd.on('request', (r) => {
+      if (!r.url().includes('/account/history/v1/checkpoint')) return;
+      try {
+        const b = JSON.parse(r.postData() ?? '{}');
+        note(`third history ${b.action}:${typeof b.payload === 'string' ? b.payload : 'object'}`);
+      } catch {}
+    });
+    Cd.on('console', (m) => {
+      if (m.type() === 'error') note('third console error: ' + m.text().slice(0, 300));
+    });
+    Cd.on('worker', (w) =>
+      w.on('console', (m) => {
+        if (m.type() === 'error') note('third worker error: ' + m.text().slice(0, 300));
+      }),
+    );
+    const online = Date.now();
+    await thirdContext.setOffline(false);
+    note('third back online');
+    const sawA = () =>
+      Cd.locator('.shell')
+        .textContent()
+        .then((t) => t!.includes('A writing the third device misses 1'));
+    const committed = await expect
+      .poll(sawA, { timeout: 90000, intervals: [200] })
+      .toBe(true)
+      .then(
+        () => true,
+        () => false,
+      );
+    const tCommit = Date.now();
+    note(
+      `install committed? ${committed} (${((tCommit - online) / 1000).toFixed(1)} s after online); revision ${await revision()}`,
+    );
+    note(
+      `B reads after returning: ${JSON.stringify(bReads.filter((r) => r.phase === 'after').slice(0, 8))}`,
+    );
+    if (!committed) {
+      note(`notice ${JSON.stringify(await sentence())}`);
+      await openMenuSection(Cd, 'Account');
+      note(`sync details ${JSON.stringify(await Cd.locator('.sync-details').allTextContents())}`);
+      await openMenuSection(Cd, 'Data');
+    }
+    {
+      const window = await waitFold(tCommit, reducedTo);
+      note(
+        `fold window: ${window} s (commit to baseline revision ${await revision()}); panel: ${JSON.stringify(await panelText())}`,
+      );
+      expect(window).toBeDefined();
+      // Exactly one fold: nothing reduces again (cooldown and growth threshold apply again).
+      const r = await revision();
+      await Cd.waitForTimeout(70000);
+      note(`70 s later revision ${await revision()} (was ${r})`);
+      expect(await revision()).toBe(r);
+      for (const older of ['tip', 'released'] as const) expect(await olderOpens(older)).toBe(true);
+    }
+    // Convergence: the three devices export the same text.
+    await add(A, 'Final writing from A');
+    for (const p of [B, Cd])
+      await expect(p.locator('.shell')).toContainText('Final writing from A', { timeout: 60000 });
+    const texts: string[] = [];
+    for (const p of [A, B, Cd]) {
+      await openMenuSection(p, 'Data');
+      const [download] = await Promise.all([
+        p.waitForEvent('download'),
+        p
+          .getByRole('region', { name: 'Data', exact: true })
+          .getByRole('button', { name: 'Export markdown', exact: true })
+          .click(),
+      ]);
+      texts.push(storedZipText(readFileSync((await download.path())!)));
+    }
+    expect(texts[1]).toBe(texts[0]);
+    expect(texts[2]).toBe(texts[0]);
+    note(`convergence: the three exports are equal (${texts[0]!.length} characters)`);
+  } finally {
+    writeFileSync(
+      info.outputPath('timeline.txt'),
+      log.join('\n') + '\n' + 'errors: ' + JSON.stringify(errors) + '\n',
+    );
+    try {
+      await thirdContext.close().catch(() => {});
+      await pair?.other.close().catch(() => {});
+    } finally {
+      await stopService(service);
+    }
+  }
+});
