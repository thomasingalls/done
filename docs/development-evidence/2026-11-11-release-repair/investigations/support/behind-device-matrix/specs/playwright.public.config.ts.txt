import { defineConfig } from '@playwright/test';
import { X509Certificate, createHash } from 'node:crypto';
import { readFileSync } from 'node:fs';
import { resolve } from 'node:path';
const cert = new X509Certificate(readFileSync(resolve('../cert/cert.pem')));
const spki = createHash('sha256')
  .update(cert.publicKey.export({ format: 'der', type: 'spki' }))
  .digest('base64');
export default defineConfig({
  testDir: './test/packaged-account',
  testMatch: /\.public\.spec\.ts$/,
  outputDir: './test-results/public/artifacts',
  timeout: 1800000,
  workers: 1,
  retries: 0,
  reporter: [['list']],
  use: {
    baseURL: 'https://localhost:53441',
    viewport: { width: 1280, height: 900 },
    launchOptions: { args: [`--ignore-certificate-errors-spki-list=${spki}`] },
  },
});
