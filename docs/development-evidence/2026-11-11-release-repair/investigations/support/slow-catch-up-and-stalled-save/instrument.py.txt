#!/usr/bin/env python3
"""COPY-ONLY trace seams for the oddities investigation. Usage: instrument.py <copy root>"""
import sys, pathlib

root = pathlib.Path(sys.argv[1])


def edit(rel, old, new, count=1):
    p = root / rel
    s = p.read_text()
    n = s.count(old)
    if n != count:
        raise SystemExit(f"{rel}: expected {count} match(es), found {n} for {old[:60]!r}")
    p.write_text(s.replace(old, new))

)
