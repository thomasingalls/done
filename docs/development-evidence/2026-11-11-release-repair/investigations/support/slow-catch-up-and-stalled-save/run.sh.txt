#!/bin/bash
# usage: run.sh <copy> <label> <spec filter> [ENV=VALUE ...]
OD=/private/tmp/claude-502/-Users-thomasingalls-Code-done--claude-worktrees-l2-onboarding-863db1/a2cb3ea1-c2ed-4b47-b08f-dc5bb9d6e506/scratchpad/oddities
copy=$1; label=$2; filter=$3; shift 3
out=$OD/runs/$label
mkdir -p $out
cd $OD/$copy || exit 1
echo "$copy $label $filter $*" > $out/params.txt
( while true; do echo "$(date +%T) $(sysctl -n vm.loadavg)"; sleep 10; done ) > $out/load.txt 2>&1 &
lp=$!
env ODD_PORT_BASE=${ODD_PORT_BASE:-54860} "$@" node node_modules/@playwright/test/cli.js test --config playwright.public.config.ts $filter > $out/out.txt 2>&1
echo "exit $?" >> $out/out.txt

echo done $label
