#!/bin/sh
# Phase 554: reduce the snapshots, then measure step 0 and F1+F1b interleaved by repetition.
set -u
SCR="$1"; REPO="$2"
cd "$REPO"
run() { # label build step [reps]
  echo "$(date '+%H:%M:%S') start $1 $3 reps=${4:-}" >> "$SCR/run-all.log"
  SEND_COST_ROOT="$SCR/sc" SEND_COST_BUILD="$SCR/builds/$2" SEND_COST_STEP="$3" SEND_COST_LABEL="$1" \
  SEND_COST_REPS="${4:-3}" SEND_COST_TIMEOUT_MS=10800000 \
    timeout 11000 pnpm exec playwright test --config scripts/performance/send-cost/playwright.config.ts \
    >> "$SCR/run-all.out" 2>&1
  echo "$(date '+%H:%M:%S') end $1 $3 exit=$?" >> "$SCR/run-all.log"
}
case "${3:-A}" in
A) run reduce step0 reduce; run step0 step0 measure 1 ;;
B) run f1 f1 measure 1; for r in 2 3; do run step0 step0 measure $r; run f1 f1 measure $r; done ;;
esac
echo "$(date '+%H:%M:%S') all done" >> "$SCR/run-all.log"
