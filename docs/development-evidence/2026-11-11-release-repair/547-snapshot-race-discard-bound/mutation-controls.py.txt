"""Phase 547 mutation controls: remove one piece, run the tests that guard it, restore the file."""

W = '<worktree>'
OUT = sys.argv[1]
ONLY = sys.argv[2:] or None

def node(*files, pattern=None):
    cmd = ['node', '--test', '--test-concurrency=1']
    if pattern:
        cmd += ['--test-name-pattern', pattern]
    return cmd + list(files)

def vitest(*files):
    return ['node', 'node_modules/vitest/vitest.mjs', 'run', '--maxWorkers=1', *files]

def summary(text):
    lines = [l for l in text.splitlines() if re.match(r'^(✖|✔|ℹ (tests|pass|fail)|\s+(×|✓)|\s*Tests\s)', l)]
    return '\n'.join(lines[-30:])

with open(OUT, 'w') as out:
    for name, path, old, new, runs in MUTATIONS:
        if ONLY and not any(name.startswith(o) for o in ONLY):
            continue
        full = f'{W}/{path}'
        src = open(full).read()
        assert src.count(old) == 1, (name, src.count(old))
        shutil.copy(full, full + '.orig')
        try:
            open(full, 'w').write(src.replace(old, new))
            out.write(f'\n## {name}\n{path}: removed\n{old}\n')
            for cmd in runs:
                p = subprocess.run(cmd, cwd=W, capture_output=True, text=True, timeout=3000)
                out.write(f'\n$ {" ".join(cmd)}\nexit {p.returncode}\n{summary(p.stdout + p.stderr)}\n')
                out.flush()
        finally:
            shutil.move(full + '.orig', full)
        assert open(full).read() == src
        print('done', name, flush=True)
