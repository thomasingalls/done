# Phase 552 base-tree run: the seven edited sources at 51779e97d, the new modules, tests and
# fixtures kept (no base source imports the new modules). Restores phase 552 from the backup.
set -u
cd <worktree>

  src/crdt/accountReceive.ts src/crdt/accountSessionBridge.ts src/crdt/accountSyncDriver.ts src/ui/SyncNotice.tsx; do
