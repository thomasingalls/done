for name, f, a, b, cmd in M:
    p = os.path.join(T, f); s = open(p).read()
    assert s.count(a) == 1, (name, a)
    open(p, 'w').write(s.replace(a, b))
    try:
        r = subprocess.run(cmd, cwd=T, capture_output=True, text=True, timeout=900)
        out = r.stdout + r.stderr
        if cmd[1] == '--test':
            failed = sorted({l.strip() for l in out.splitlines() if l.startswith('✖ ') and 'failing tests' not in l})
            summary = [l.strip() for l in out.splitlines() if l.startswith('ℹ fail') or l.startswith('ℹ pass')]
        else:
            failed = [l.strip() for l in out.splitlines() if l.strip().startswith('×')]
            summary = [l.strip() for l in out.splitlines() if l.strip().startswith('Tests ')]
        print(f'{name}: exit {r.returncode}; {"; ".join(summary[-2:]) if summary else "?"}', flush=True)
        for l in failed: print('   ', l[:170], flush=True)
    finally:
        open(p, 'w').write(s)
