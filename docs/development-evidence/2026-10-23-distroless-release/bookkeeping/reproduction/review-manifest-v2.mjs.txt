let caseArrays = 0;
function equalContract(before, after, path) {
  if (Array.isArray(before)) {
    if (path.endsWith('.cases')) {
      caseArrays++;
      const original = before.map((c) => ({ ...c, id: rename(c.id) }));
      const ids = new Set(original.map((c) => c.id));
      assert.equal(new Set(after.map((c) => c.id)).size, after.length, path);
      assert.deepEqual(after.filter((c) => ids.has(c.id)), original, path);
      return;
    }
    assert(Array.isArray(after)); assert.equal(after.length, before.length, path);
    before.forEach((value, i) => equalContract(value, after[i], path + '[' + i + ']'));
  } else if (before && typeof before === 'object') {
    assert.deepEqual(Object.keys(after).sort(), Object.keys(before).sort(), path);
    for (const key of Object.keys(before)) equalContract(before[key], after[key], path + '.' + key);
  } else assert.deepEqual(after, before, path);
}
