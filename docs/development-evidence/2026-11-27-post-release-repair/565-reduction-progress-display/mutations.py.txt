def sha(path):
    return hashlib.sha256(open(path, 'rb').read()).hexdigest()

def run(spec, tag):
    out = os.path.join(HERE, f'vitest-{tag}.json')
    if os.path.exists(out):
        os.remove(out)
    cmd = ['pnpm', 'exec', 'vitest', 'run', *spec, '--reporter=json', f'--outputFile={out}']
    began = time.time()
    try:
        subprocess.run(cmd, capture_output=True, text=True, timeout=400)
    except subprocess.TimeoutExpired:
        return {'error': 'timeout'}
    if not os.path.exists(out):
        return {'error': 'no report (probably a compile failure)', 'seconds': round(time.time() - began, 1)}
    data = json.load(open(out))
    cases = [a for r in data['testResults'] for a in r['assertionResults']]
    return {
        'seconds': round(time.time() - began, 1),
        'passed': sum(1 for a in cases if a['status'] == 'passed'),
        'failed': [a['fullName'] for a in cases if a['status'] == 'failed'],
        'skipped': sum(1 for a in cases if a['status'] in ('skipped', 'pending', 'todo')),
    }

results = {'asBuilt': {}, 'mutations': []}
for name, spec in [('estimator', EST), ('panel', PANEL), ('plumbing', PLUMB)]:
    results['asBuilt'][name] = run(spec, 'built-' + name)
for mid, desc, path, edits, spec in M:
    original = open(path).read()
    digest = sha(path)
    mutated = original
    for old, new in edits:
        if mutated.count(old) != 1:
            print('cannot apply', mid, repr(old[:60]), file=sys.stderr)
            sys.exit(2)
        mutated = mutated.replace(old, new)
    try:
        open(path, 'w').write(mutated)
        r = run(spec, mid)
    finally:
        open(path, 'w').write(original)
    restored = sha(path) == digest
    results['mutations'].append({'id': mid, 'variant': desc, 'file': path, 'tests': spec[0],
                                 'restored': restored, **r})
    print(mid, 'failed', len(r.get('failed', [])), r.get('error', ''), 'restored', restored, flush=True)
    if not restored:
        sys.exit(3)
json.dump(results, open(os.path.join(HERE, 'mutations-result.json'), 'w'), indent=1)
