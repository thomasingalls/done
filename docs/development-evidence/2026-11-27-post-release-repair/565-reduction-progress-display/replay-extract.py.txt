import gzip, re, sys, ast, json
def parse(path):
    text = gzip.open(path, 'rt').read()
    rows = []
    for line in text.splitlines():
        m = re.match(r'^\s*(\d+)\s+(-?[\d.]+)\s+([Yn])\s+(-?[\d.]+)\s+(.*)$', line)
        if not m: continue
        n, start, deff, prot, rest = m.groups()
        toks = rest.split()
        # trailing: budget setup imp hits/miss save end result
        result = toks[-1]; end = float(toks[-2]); save = float(toks[-3]); hm = toks[-4]; imp = toks[-5]; setup = float(toks[-6]); budget = float(toks[-7])
        stages = toks[:-7]
        st = []
        for s in stages:
            mm = re.match(r'(\w+):(-?[\d.]+)(?:\[(\w+)/(\d+)\])?', s)
            name, dur, done, total = mm.groups()
            st.append((name, float(dur), None if done in (None,'None') else int(done), None if total is None else int(total)))
        hits, miss = (-1, -1) if hm == '-' else map(int, hm.split('/'))
        rows.append(dict(n=int(n), start=float(start), prot=prot, stages=st, budget=budget, setup=setup,
                         imp=-1 if imp=='-' else int(imp), hits=hits, miss=miss, save=save, end=end, result=result))
    bl = re.search(r'budgets (\[.*\])', text)
    budgets = ast.literal_eval(bl.group(1)) if bl else []
    return rows, budgets
for p in sys.argv[1:]:
    rows, budgets = parse(p)
    withveri = [r for r in rows if any(s[0]=='veri' for s in r['stages'])]
    withmiss = [r for r in rows if r['miss'] > 0]
    print(p, 'rows', len(rows), 'veri', len(withveri), 'miss>0', len(withmiss), 'budgets', len(budgets))
    print(' results', {k: sum(1 for r in rows if r['result']==k) for k in set(r['result'] for r in rows)})

def emit(path, name):
    rows, budgets = parse(path)
    out = []
    bi = 0
    for r in rows:
        st = r['stages']
        names = [s[0] for s in st]
        before = 0.0; opened = (0, 0); veri = -1
        for s in st:
            if s[0] == 'veri': veri = round(s[1]*1000); break
            before += s[1]
            if s[0] == 'open': opened = (s[2] or 0, s[3] or 0)
        budget = setup = -1
        if veri >= 0:
            b = budgets[bi]; bi += 1
            budget, setup = b[0], b[1]
        outcome = {'preparation-saved/None': 0, 'removed/None': 2}.get(r['result'], 1)
        build = store = 0
        if outcome == 2:
            after = st[names.index('veri')+1:]
            build = round(sum(s[1] for s in after if s[0] in ('buil','cove','curr'))*1000)
            store = round(sum(s[1] for s in after if s[0] in ('inst','read','comp'))*1000)
        out.append([round(r['start']*1000), round(before*1000), opened[0], veri, budget, setup, r['imp'], r['miss'], round(r['save']*1000), round(r['end']*1000), outcome, build, store])
    print(f'// {name}: {len(out)} attempts')
    print(f'const {name}: Attempt[] = [')
    for o in out: print('  [' + ', '.join(str(x) for x in o) + '],')
    print('];')
import os
base = os.environ.get('RUNS', '.')
emit(os.path.join(base, 'run-r06-1000-ts100-offline.log.gz'), 'r06')
emit(os.path.join(base, 'run-r07-1000-ts20-online.log.gz'), 'r07')
