function pageProbe() {
  const Real = BroadcastChannel;
  (window as unknown as { BroadcastChannel: unknown }).BroadcastChannel = class extends Real {
    constructor(n: string) {
      super(n);
      if (n === 'done/slot-lifecycle')
        this.addEventListener('message', (e) =>
          console.log('[P] bc message', JSON.stringify((e as MessageEvent).data)),
        );
    }
  };
}
const screen = async (page: Page) =>
  JSON.stringify({
    h1: await page.locator('h1').allTextContents(),
    status: await page.locator('[role=status]').allTextContents(),
    alert: await page.locator('[role=alert]').allTextContents(),
    workers: page.workers().map((w) => w.url().match(/(\w+)\.worker/)?.[1]),
  });

for (const mode of ['held-begin', 'idle-passkey'] as const)
  test(`phase 225 cooperative ${mode}`, async ({ page, context }, info) => {
    const service = await startService(info.outputPath('service'), undefined, true, true);
    const lines: string[] = [];
