function suppressRetirement() {
  const Real = BroadcastChannel;
  let suppressed = 0;
  (self as unknown as { BroadcastChannel: unknown }).BroadcastChannel = class extends Real {
    constructor(name: string) {
      super(name);
      if (name === 'done/slot-lifecycle')
        this.addEventListener('message', (event) => {
          suppressed++;
          event.stopImmediatePropagation();
        });
    }
  };
  Object.defineProperty(self, '__suppressedRetirements', { get: () => suppressed });
}
const screen = async (page: Page) =>
  JSON.stringify({
    h1: await page.locator('h1').allTextContents(),
    status: await page.locator('[role=status]').allTextContents(),
    alert: await page.locator('[role=alert]').allTextContents(),
  });

for (const fresh of [false, true])
  test(`missed notice in the Worker${fresh ? ' after Start fresh' : ''}`, async ({ page, context }, info) => {
