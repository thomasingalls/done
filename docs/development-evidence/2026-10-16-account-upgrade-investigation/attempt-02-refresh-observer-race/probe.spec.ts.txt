async function processStart(source: string, script: string, ready: string, dist: string, phase: string) {
  const child = spawn(process.execPath, [resolve(source, script)], {
    cwd: source, stdio: ['ignore', 'pipe', 'pipe'], env: {
      ...process.env, ACCOUNT_SETUP_PORT: '53221', ACCOUNT_SERVICE_PORT: '53222',
      ACCOUNT_SETUP_ORIGIN: origin, ACCOUNT_SETUP_STATE: output + '/service',
      ACCOUNT_SETUP_DIST: dist, ACCOUNT_NETWORK_SESSIONS: '1', ACCOUNT_MEMBERSHIP_CONTINUATION: '1',
    },
  });
  let log = '';
  child.stdout!.on('data', data => { log += String(data); writeFileSync(output + '/' + phase + '-' + script.split('/').at(-1) + '.log', log); });
  child.stderr!.on('data', data => { log += String(data); writeFileSync(output + '/' + phase + '-' + script.split('/').at(-1) + '.log', log); });
  await new Promise<void>((yes, no) => {
    const timeout = setTimeout(() => { child.kill('SIGTERM'); no(Error('Test service startup timeout: ' + log.slice(-1500))); }, 15000);
    const check = () => { if (log.includes(ready)) { clearTimeout(timeout); yes(); } };
    child.stdout!.on('data', check); child.stderr!.on('data', check);
    child.once('error', error => { clearTimeout(timeout); no(error); });
    child.once('exit', code => { clearTimeout(timeout); no(Error('Test service exited ' + code + ': ' + log.slice(-1500))); });
  });
  return child;
}
async function stop(child: ChildProcess | undefined) {
  if (!child || child.exitCode !== null || child.signalCode !== null) return;
  await new Promise<void>((yes, no) => {
    const timeout = setTimeout(() => { child.kill('SIGKILL'); no(Error('Test service did not drain')); }, 10000);
    child.once('exit', () => { clearTimeout(timeout); yes(); }); child.kill('SIGTERM');
  });
}
async function service(source: string, dist: string, phase: string) {
  const api = await processStart(source, 'scripts/serve-account-service.mjs', 'Account API development:', dist, phase);
  try {
    const app = await processStart(source, 'scripts/serve-account.mjs', 'Account packaged development:', dist, phase);
    return { api, app };
  } catch (error) { await stop(api); throw error; }
}
async function instrument(context: BrowserContext, browser: string) {
  await context.exposeBinding('__incidentEvent', (_source, detail) => { events.push({ browser, ...detail }); });
  await context.addInitScript(() => {
    const emit = (detail: Record<string, unknown>) => {
      const binding = (window as unknown as {__incidentEvent?: (detail: unknown) => Promise<void>}).__incidentEvent;
      void binding?.({ at: Date.now(), path: location.pathname, ...detail }).catch(() => {});
    };
    const open = IDBFactory.prototype.open;
    IDBFactory.prototype.open = function(name, version) {
      const request = version === undefined ? open.call(this, name) : open.call(this, name, version);
      if (name === 'done' || name.startsWith('done.')) request.addEventListener('upgradeneeded', event => emit({ kind: 'idb-upgrade', database: name, oldVersion: event.oldVersion, newVersion: event.newVersion }));
      return request;
    };
    const deleteDatabase = IDBFactory.prototype.deleteDatabase;
    IDBFactory.prototype.deleteDatabase = function(name) {
      if (name === 'done' || name.startsWith('done.')) emit({ kind: 'idb-delete-database', database: name });
      return deleteDatabase.call(this, name);
    };
    for (const method of ['setItem', 'removeItem', 'clear'] as const) {
      const original = Storage.prototype[method];
      Object.defineProperty(Storage.prototype, method, { configurable: true, writable: true, value: function(this: Storage, ...args: string[]) {
        const key = args[0];
        const result = Reflect.apply(original, this, args);
        if (method === 'clear' || key?.startsWith('done.')) emit({ kind: 'storage-' + method, storage: this === localStorage ? 'local' : 'session', key: method === 'clear' ? null : key });
        return result;
      } });
    }
    for (const method of ['put', 'add', 'delete', 'clear'] as const) {
      const original = IDBObjectStore.prototype[method];
      Object.defineProperty(IDBObjectStore.prototype, method, { configurable: true, writable: true, value: function(this: IDBObjectStore, ...args: unknown[]) {
        const database = this.transaction.db.name, store = this.name;
        const result = Reflect.apply(original, this, args);
        if (['done.slots', 'done.custody', 'done.first-device-setup.v1', 'done.account-login.v1'].includes(database)) {
          emit({ kind: 'idb-' + method, database, store, phase: 'requested' });
          this.transaction.addEventListener('complete', () => emit({ kind: 'idb-' + method, database, store, phase: 'committed' }), { once: true });
          this.transaction.addEventListener('abort', () => emit({ kind: 'idb-' + method, database, store, phase: 'aborted' }), { once: true });
        }
        return result;
      } });
    }
    navigator.serviceWorker?.addEventListener('controllerchange', () => emit({kind: 'sw-controllerchange', controller: navigator.serviceWorker.controller?.scriptURL}));
    window.addEventListener('pageshow', event => emit({kind: 'pageshow', persisted: event.persisted}));
    window.addEventListener('pagehide', event => emit({kind: 'pagehide', persisted: event.persisted}));
    emit({kind: 'document-start'});
  });
}
async function snapshot(page: Page, browser: string, stage: string) {
  const state = await page.evaluate(async () => {
    const stable = (value: unknown): unknown => {
      if (value instanceof CryptoKey) return {kind:'CryptoKey',algorithm:value.algorithm,type:value.type,extractable:value.extractable,usages:value.usages};
      if (value instanceof ArrayBuffer) return {kind:'bytes',value:Array.from(new Uint8Array(value))};
      if (ArrayBuffer.isView(value)) return {kind:'bytes',value:Array.from(new Uint8Array(value.buffer,value.byteOffset,value.byteLength))};
      if (Array.isArray(value)) return value.map(stable);
      if (value && typeof value === 'object') return Object.fromEntries(Object.entries(value).sort(([a],[b])=>a.localeCompare(b)).map(([k,v])=>[k,stable(v)]));
      return value === undefined ? {kind:'undefined'} : value;
    };
    const digest = async (value: unknown) => Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256',new TextEncoder().encode(JSON.stringify(stable(value))))), b=>b.toString(16).padStart(2,'0')).join('');
    const databases: {name: string;version: number;stores: Record<string,{count:number;keysDigest:string;rowsDigest:string}>}[] = [];
    for (const info of (await indexedDB.databases()).filter(d=>d.name==='done'||d.name?.startsWith('done.'))) {
      const db = await new Promise<IDBDatabase>((yes,no)=>{
        const r=indexedDB.open(info.name!); r.onupgradeneeded=()=>r.transaction!.abort();
        r.onsuccess=()=>yes(r.result);r.onerror=()=>no(r.error);r.onblocked=()=>no(Error('Snapshot blocked'));
      });
      try {
        const stores: Record<string,{count:number;keysDigest:string;rowsDigest:string}> = {};
        for(const name of db.objectStoreNames) {
          const raw=await new Promise<{keys:IDBValidKey[];values:unknown[]}>((yes,no)=>{
            const tx=db.transaction(name,'readonly'),store=tx.objectStore(name),keys=store.getAllKeys(),values=store.getAll();
            tx.oncomplete=()=>yes({keys:keys.result,values:values.result});tx.onabort=()=>no(tx.error);
          });
          stores[name]={count:raw.keys.length,keysDigest:await digest(raw.keys),rowsDigest:await digest(raw)};
        }
        databases.push({name:db.name,version:db.version,stores});
      } finally { db.close(); }
    }
    const local: Record<string,string>={};
    for(const key of Object.keys(localStorage).filter(k=>k.startsWith('done.')).sort()) local[key]=await digest(localStorage.getItem(key));
    const registrations=await navigator.serviceWorker.getRegistrations();
    return {
      origin:location.origin,path:location.pathname,mainScript:document.querySelector<HTMLScriptElement>('script[type=module][src]')?.getAttribute('src'),
      databases:databases.sort((a,b)=>a.name.localeCompare(b.name)),local,
      serviceWorkers:registrations.map(r=>({scope:r.scope,active:r.active?.state,waiting:r.waiting?.state,installing:r.installing?.state})),
      controlled:!!navigator.serviceWorker.controller,cacheNames:await caches.keys(),
      headings:Array.from(document.querySelectorAll('h1,h2')).map(e=>e.textContent),
      buttons:Array.from(document.querySelectorAll('button')).map(e=>e.textContent?.trim()).filter(Boolean),
      editableApp:!!document.querySelector('.shell'),
    };
  });
  const record={browser,stage,at:Date.now(),...state}; observations.push(record);persist();return record;
}
